# ⚡ Электроника

```dataviewjs
/* ═══ Хаб раздела «Электроника» ═══ */
let DIR = dv.current().file.folder, SRC = null, err = null;
while (DIR) {
    try { SRC = await app.vault.adapter.read(DIR + "/⚙️ движок.js"); break; } catch (e) { err = e; }
    const i = DIR.lastIndexOf("/"); DIR = i < 0 ? null : DIR.slice(0, i);
}
if (!SRC) { dv.paragraph("⚠️ Не найден движок раздела «Электроника». Ошибка: " + (err && err.message || "")); }
else {
    const elMain = new Function("dv", "app", SRC + "\nreturn elMain;")();
    await elMain(dv, app, "hub");
}
```
