const esc = s => String(s == null ? "" : s).replace(/[&<>"']/g, c => ({ "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#39;" }[c]));

/* инлайн-форматирование: жирный (**), курсив (* или _), код (`), зачёркнутый (~~), "кавычки"→«», ...→… */
function mdInline(s) {
  let t = String(s == null ? "" : s)
    .replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
  t = t.replace(/"([^"\n]+)"/g, "«$1»");
  t = t.replace(/\.\.\./g, "…");
  t = t.replace(/`([^`\n]+)`/g, "<code>$1</code>");
  t = t.replace(/\*\*([^*\n]+)\*\*/g, "<strong>$1</strong>");
  t = t.replace(/__([^_\n]+)__/g, "<strong>$1</strong>");
  t = t.replace(/(^|[^*])\*([^*\n]+)\*/g, "$1<em>$2</em>");
  t = t.replace(/(^|[^_\w])_([^_\n]+)_/g, "$1<em>$2</em>");
  t = t.replace(/~~([^~\n]+)~~/g, "<s>$1</s>");
  return t;
}

function parseAmount(amt) {
  const m = String(amt).match(/^(\d+\s*\/\s*\d+|\d+(?:[.,]\d+)?)\s*(.*)$/);
  if (!m) return { qty: null, unit: "", note: String(amt) };
  const q = m[1].replace(/\s+/g, "");
  let qty;
  if (q.includes("/")) { const [a, b] = q.split("/").map(Number); qty = b ? a / b : null; }
  else qty = parseFloat(q.replace(",", "."));
  return { qty, unit: m[2].trim(), note: "" };
}
function parseRecipe(raw) {
  const lines = String(raw).split(/\r?\n/);
  const ings = [], steps = [];
  let mode = null;
  for (const line of lines) {
    const h = line.match(/^#{1,6}\s+(.*)$/);
    if (h) {
      const t = h[1].toLowerCase();
      if (/ингредиент/.test(t)) mode = "ing";
      else if (/приготовл|инструкц|процесс|шаг/.test(t)) mode = "steps";
      else mode = null;
      continue;
    }
    if (mode === "ing") {
      const m = line.match(/^\s*[-*+]\s+(.+?)\s*$/);
      if (m) {
        const body = m[1];
        const sep = body.match(/^(.*?)\s*[—–]\s*(.+)$/);
        let name = body, amt = "";
        if (sep) { name = sep[1].trim(); amt = sep[2].trim(); }
        const a = amt ? parseAmount(amt) : { qty: null, unit: "", note: "" };
        ings.push({ name, qty: a.qty, unit: a.unit, note: a.note });
      }
    } else if (mode === "steps") {
      const m = line.match(/^\s*(?:\d+[.)]|[-*+])\s+(.+?)\s*$/);
      if (m) {
        let text = m[1], min = null;
        const tm = text.match(/⏱\s*([\d.,]+)\s*(ч\w*)?/i);
        if (tm) {
          let v = parseFloat(tm[1].replace(",", "."));
          if (tm[2] && /^ч/i.test(tm[2])) v *= 60;
          min = v;
          text = text.replace(tm[0], "").replace(/\s{2,}/g, " ").replace(/\s+([.,;:])/g, "$1").trim();
        }
        steps.push({ text, min });
      }
    }
  }
  return { ings, steps };
}

async function loadRecipes() {
  const pages = dv.pages()
    .where(p => p.type === "рецепт"
      && p.file.path !== TPL
      && !/^Templates\//.test(p.file.path)
      && !String(p.title || "").includes("{{"))
    .array();
  const out = [];
  for (const p of pages) {
    let raw = "";
    try {
      const f = app.vault.getAbstractFileByPath(p.file.path);
      if (f) raw = await app.vault.read(f);
    } catch (e) {}
    const parsed = parseRecipe(raw);
    const title = String(p.title || p.file.name);
    const cat = String(p.category || "Без категории");
    let mt = 0;
    try { mt = p.file.mtime && p.file.mtime.toMillis ? p.file.mtime.toMillis() : Number(p.file.mtime) || 0; } catch (e) {}
    out.push({
      path: p.file.path,
      title, category: cat,
      emoji: String(p.emoji || catEmoji(cat, title)),
      servings: Number(p.servings) || 0,
      time: Number(p.time) || 0,
      difficulty: String(p.difficulty || ""),
      kcal: p.kcal ? String(p.kcal) : "",
      tags: Array.isArray(p.tags) ? p.tags.map(String) : (p.tags ? [String(p.tags)] : []),
      mtime: mt,
      ings: parsed.ings, steps: parsed.steps,
    });
  }
  out.sort((a, b) => a.title.localeCompare(b.title, "ru"));
  return out;
}

function beep() {
  try {
    const AC = window.AudioContext || window.webkitAudioContext;
    const ctx = (window.__rcAudio = window.__rcAudio || new AC());
    if (ctx.state === "suspended" && ctx.resume) ctx.resume();
    const t0 = ctx.currentTime;
    [0, 0.28, 0.56].forEach((d, i) => {
      const o = ctx.createOscillator(), g = ctx.createGain();
      o.type = "sine"; o.frequency.value = i === 2 ? 1180 : 880;
      o.connect(g); g.connect(ctx.destination);
      g.gain.setValueAtTime(0.0001, t0 + d);
      g.gain.exponentialRampToValueAtTime(0.28, t0 + d + 0.02);
      g.gain.exponentialRampToValueAtTime(0.0001, t0 + d + 0.24);
      o.start(t0 + d); o.stop(t0 + d + 0.26);
    });
  } catch (e) {}
}

        </select>
      </div>
      <div class="rb-modal-btns">
        <button class="rb-btn-ghost" data-x="cancel">Отмена</button>
        <button class="rb-btn-primary" data-x="ok">Создать</button>
      </div>
    </div>`;
  document.body.appendChild(m);
  const get = f => m.querySelector('[data-f="' + f + '"]').value;
  const titleI = m.querySelector('[data-f="title"]');
  setTimeout(() => titleI.focus(), 30);

  function closeModal() { m.remove(); document.removeEventListener("keydown", onKeyM, true); }
  async function submit() {
    const name = (get("title") || "").trim();
    if (!name) { new Notice("Введите название рецепта"); titleI.focus(); return; }
    const safe = name.replace(/[\\/:*?"<>|#^\[\]]/g, " ").replace(/\s+/g, " ").trim();
    const path = DIR + "/" + safe + ".md";
    if (app.vault.getAbstractFileByPath(path)) { new Notice("Такой рецепт уже есть"); return; }
    let tpl = "";
    try { const tf = app.vault.getAbstractFileByPath(TPL); if (tf) tpl = await app.vault.read(tf); } catch (e) {}
    if (!tpl) tpl = DEFAULT_TPL;
    const clean = s => String(s == null ? "" : s).replace(/"/g, "'").trim();
    const content = tpl
      .replace(/\{\{TITLE\}\}/g, clean(name))
      .replace(/\{\{CATEGORY\}\}/g, clean(get("category")) || "Без категории")
      .replace(/\{\{SERVINGS\}\}/g, String(parseInt(get("servings"), 10) || 2))
      .replace(/\{\{TIME\}\}/g, String(parseInt(get("time"), 10) || 30))
      .replace(/\{\{DIFFICULTY\}\}/g, clean(get("difficulty")) || "средне");
    try {
      const f = await app.vault.create(path, content);
      closeModal();
      await app.workspace.getLeaf(false).openFile(f);
      new Notice("✅ Рецепт создан — заполни ингредиенты и шаги");
      if (onCreated) onCreated();
    } catch (e) { new Notice("Ошибка создания: " + e); }
  }
  function onKeyM(e) {
    if (e.key === "Escape") { e.preventDefault(); closeModal(); }
    else if (e.key === "Enter") {
      const tg = e.target && e.target.tagName;
      if (tg === "BUTTON" || tg === "SELECT") return; // не мешаем кнопкам
      e.preventDefault(); submit();
    }
  }
  document.addEventListener("keydown", onKeyM, true);
  m.addEventListener("click", e => {
    if (e.target === m) { closeModal(); return; }
    const x = e.target.closest("[data-x]");
    if (!x) return;
    if (x.getAttribute("data-x") === "ok") submit(); else closeModal();
  });
}

function plural(n, a, b, c) { const m10 = n % 10, m100 = n % 100; if (m10 === 1 && m100 !== 11) return a; if (m10 >= 2 && m10 <= 4 && (m100 < 10 || m100 >= 20)) return b; return c; }

function matches(r) {
  if (cat !== "all" && r.category !== cat) return false;
  if (!q) return true;
  const hay = (r.title + " " + r.category + " " + r.tags.join(" ") + " " + r.ings.map(i => i.name).join(" ")).toLowerCase();
  return q.split(/\s+/).every(w => hay.includes(w));
}

  if (sortMode === "title") list = list.slice().sort((a, b) => a.title.localeCompare(b.title, "ru"));
