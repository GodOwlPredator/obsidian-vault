# 🎨 Выделители

Плавающая палитра «как маркер в жизни»: выделяешь текст в любой заметке, жмёшь цвет — и фрагмент окрашивается рваным мазком. Цвет — любой из RGB (с прозрачностью), форма мазка настраивается. Всё применяется на лету и к уже размеченным выделениям.

function wrapSelected(insert) {
  const ctx = app.workspace.activeEditor;
  const ed = ctx && (ctx.editor || (ctx.view && ctx.view.editor));
  if (!ed) { new Notice("Открой заметку с текстом"); return; }
  const cm = ed.cm || ed;
  if (!cm.state || !cm.state.selection) { new Notice("Не удалось получить редактор"); return; }
  const sel = cm.state.selection.main;
  if (sel.from === sel.to) { new Notice("Сначала выдели текст мышкой"); return; }
  const txt = cm.state.doc.sliceString(sel.from, sel.to);
  const re = /^<mark class="hl-[^"]*">([\s\S]*)<\/mark>$/;
  const out = insert === null
    ? (re.test(txt) ? txt.replace(re, "$1") : txt)
    : insert.replace("###TXT###", txt);
  cm.dispatch({ changes: { from: sel.from, to: sel.to, insert: out } });
}

const hkInps = {};
function fmtHk(hk) {
  if (!hk || (!hk.key && !hk.code)) return "";
  const p = [];
  if (hk.ctrl)  p.push("Ctrl");
  if (hk.alt)   p.push("Alt");
  if (hk.shift) p.push("Shift");
  let k;
  if (hk.code) {
    if     (hk.code.startsWith("Key"))    k = hk.code.slice(3);
    else if (hk.code.startsWith("Digit"))  k = hk.code.slice(5);
    else if (hk.code.startsWith("Numpad")) k = "Num" + hk.code.slice(6);
    else                                   k = hk.code;
  } else {
    k = hk.key === " " ? "Space" : hk.key.length === 1 ? hk.key.toUpperCase() : hk.key;
  }
  p.push(k);
  return p.join(" + ");
}
function matchHk(e, hk) {
  if (!hk || (!hk.key && !hk.code)) return false;
  const keyMatch = hk.code ? e.code === hk.code : e.key.toLowerCase() === hk.key;
  return keyMatch && e.ctrlKey === !!hk.ctrl && e.shiftKey === !!hk.shift && e.altKey === !!hk.alt;
}
function bindHk(inp, cfgKey) {
  inp.addEventListener("click", () => {
    inp.value = ""; inp.placeholder = "Нажми комбинацию…"; inp.classList.add("cap");
    inp.focus();
    const handler = (e) => {
      if (["Shift", "Control", "Alt", "Meta"].includes(e.key)) return;
      e.preventDefault(); e.stopPropagation();
      if (e.key === "Escape") {
        inp.classList.remove("cap"); syncHk();
        document.removeEventListener("keydown", handler, true);
        return;
      }
      cfg.hotkeys[cfgKey] = { key: e.key.toLowerCase(), code: e.code, ctrl: e.ctrlKey, shift: e.shiftKey, alt: e.altKey };
      save();
      inp.value = fmtHk(cfg.hotkeys[cfgKey]); inp.placeholder = ""; inp.classList.remove("cap");
      document.removeEventListener("keydown", handler, true);
    };
    document.addEventListener("keydown", handler, true);
  });
}
