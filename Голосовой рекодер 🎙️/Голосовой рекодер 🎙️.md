// ── Утилиты хоткеев ───────────────────────────────────────────
function fmtHk(hk){
    if(!hk?.key && !hk?.code) return "";
    const p=[];
    if(hk.ctrl)  p.push("Ctrl");
    if(hk.alt)   p.push("Alt");
    if(hk.shift) p.push("Shift");
    let k;
    if(hk.code){
        // Читаемое имя из физического кода — не зависит от раскладки
        if     (hk.code.startsWith("Key"))    k = hk.code.slice(3);           // KeyR → R
        else if(hk.code.startsWith("Digit"))  k = hk.code.slice(5);           // Digit1 → 1
        else if(hk.code.startsWith("Numpad")) k = "Num" + hk.code.slice(6);   // Numpad1 → Num1
        else                                  k = hk.code;                     // F1, Space, Enter…
    } else {
        // Обратная совместимость со старыми сохранёнными хоткеями без поля code
        k = hk.key===" " ? "Space" : hk.key.length===1 ? hk.key.toUpperCase() : hk.key;
    }
    p.push(k);
    return p.join(" + ");
}
function matchHk(e, hk){
    if(!hk?.key && !hk?.code) return false;
    // e.code — физическая клавиша, не зависит от раскладки (как в играх)
    const keyMatch = hk.code
        ? e.code === hk.code
        : e.key.toLowerCase() === hk.key;   // fallback для старых записей
    return keyMatch &&
           e.ctrlKey===!!hk.ctrl &&
           e.shiftKey===!!hk.shift &&
           e.altKey===!!hk.alt;
}

// ── Захват хоткея: клик на поле → следующее нажатие ──────────
function bindCapture(inp, cfgKey){
    inp.addEventListener("click",()=>{
        inp.value=""; inp.placeholder="Нажми комбинацию…"; inp.classList.add("cap");
        inp.focus();
        const handler=e=>{
            // Пропускаем одиночные модификаторы
            if(["Shift","Control","Alt","Meta"].includes(e.key)) return;
            e.preventDefault(); e.stopPropagation();
            if(e.key==="Escape"){
                inp.classList.remove("cap"); syncCfgFields();
                document.removeEventListener("keydown",handler,true); return;
            }
            CFG[cfgKey]={ key:e.key.toLowerCase(), code:e.code, ctrl:e.ctrlKey, shift:e.shiftKey, alt:e.altKey };
            inp.value=fmtHk(CFG[cfgKey]); inp.placeholder=""; inp.classList.remove("cap");
            document.removeEventListener("keydown",handler,true);
        };
        document.addEventListener("keydown",handler,true);
    });
}

function drawIdle(){
    const W=canvasEl.offsetWidth||246, H=canvasEl.offsetHeight||44;
    ctx.clearRect(0,0,W,H);
    ctx.strokeStyle="#2a2a2a"; ctx.lineWidth=1.5;
    ctx.beginPath(); ctx.moveTo(0,H/2); ctx.lineTo(W,H/2); ctx.stroke();
}

function drawLive(){
    if(!analyser) return;
    analyser.getByteFrequencyData(freqData);
    const W=canvasEl.offsetWidth||246, H=canvasEl.offsetHeight||44;
    ctx.clearRect(0,0,W,H);
    const count=freqData.length, barW=(W-(count-1)*2)/count, maxH=H-6;
    for(let i=0;i<count;i++){
        const v=freqData[i]/255, bh=Math.max(2,v*maxH), x=i*(barW+2), y=(H-bh)/2;
        const grad=ctx.createLinearGradient(0,y+bh,0,y);
        grad.addColorStop(0,C.a+"55"); grad.addColorStop(1,C.a);
        ctx.fillStyle=grad;
        ctx.beginPath(); ctx.roundRect(x,y,barW,bh,2); ctx.fill();
    }
    rafId=requestAnimationFrame(drawLive);
}
function stopDraw(){ if(rafId){ cancelAnimationFrame(rafId); rafId=null; } drawIdle(); }

        for(let i=0;i<e.results.length;i++){
            if(e.results[i].isFinal) finalTxt   +=e.results[i][0].transcript+" ";
            else                      interimTxt +=e.results[i][0].transcript;
        }
        const full=(finalTxt+interimTxt).trim();
        if(full){
            tBox.textContent=full;
            tBox.className=interimTxt?"show itr":"show";
            tIns.classList.add("show");
        }
    };
    recog.onerror=()=>{}; // тихо игнорируем (микрофон уже занят MediaRecorder)
    try{ recog.start(); }catch(e){}
}

async function startRec(){
    recBtn.disabled=true; // блокируем двойной клик пока ждём разрешения
    try{
        // Максимальное качество для голоса — как в мессенджерах
        const stream=await navigator.mediaDevices.getUserMedia({
            audio:{
                echoCancellation:  true,   // убираем эхо
                noiseSuppression:  true,   // шумоподавление
                autoGainControl:   true,   // автоусиление
                sampleRate:        48000,  // 48 кГц — стандарт Opus/WebRTC
                channelCount:      1,      // моно для голоса
            }
        });
        chunks=[]; finalTxt=""; interimTxt="";
        tBox.textContent=""; tBox.className=""; tIns.classList.remove("show");

        // Web Audio API для отрисовки реальной волны
        const audioCtx=new AudioContext();
        analyser=audioCtx.createAnalyser();
        analyser.fftSize=64;
        analyser.smoothingTimeConstant=0.75;
        freqData=new Uint8Array(analyser.frequencyBinCount);
        audioCtx.createMediaStreamSource(stream).connect(analyser);

        // Предпочитаем Opus — лучший кодек для голоса (используется в TG, Discord, WebRTC)
        const mime=['audio/webm;codecs=opus','audio/webm']
            .find(t=>MediaRecorder.isTypeSupported(t))||'';
        mediaRec=new MediaRecorder(stream, mime?{mimeType:mime}:{});
        mediaRec.ondataavailable=e=>{ if(e.data.size>0) chunks.push(e.data); };
        // Передаём mime в saveAudio чтобы blob создался с правильным типом
        mediaRec.onstop=()=>{ audioCtx.close(); saveAudio(stream, mime||'audio/webm'); };
        // Без timeslice: все данные приходят одним куском при stop()
        // Это исправляет ошибку при коротких записях (1-2 сек)
        mediaRec.start();

        ts=Date.now();
        tick=setInterval(()=>{ timerEl.textContent=fmt(Date.now()-ts); },500);
        startSR();
        rafId=requestAnimationFrame(drawLive);

        recBtn.disabled=false; // разрешаем нажатие (теперь для остановки)
        recBtn.textContent="Остановить запись"; recBtn.classList.add("on");
        timerEl.classList.add("on"); dotEl.classList.add("on"); el.classList.add("is-rec");
        setStatus("Идёт запись…");
    }catch(err){
        recBtn.disabled=false; // разблокируем если не удалось запустить
        setStatus("Нет доступа к микрофону","err");
    }
}

async function saveAudio(stream, mimeType='audio/webm'){
    stream.getTracks().forEach(t=>t.stop());
    if(recog){ try{ recog.stop(); }catch(e){} }
    clearInterval(tick);
    stopDraw(); analyser=null; freqData=null;

    timerEl.textContent="00:00"; timerEl.classList.remove("on");
    recBtn.textContent="Начать запись"; recBtn.classList.remove("on");
    dotEl.classList.remove("on"); el.classList.remove("is-rec");

    const blob=new Blob(chunks,{type:mimeType});

    // Защита от пустой / слишком короткой записи
    if(blob.size < 500){
        setStatus("Запись пустая — попробуй ещё раз","err");
        return;
    }

    const now=new Date();
    const name=`Голосовая ${pad2(now.getDate())}-${pad2(now.getMonth()+1)}-${now.getFullYear()} ${pad2(now.getHours())}-${pad2(now.getMinutes())}`;
    const file=`${name}.webm`;
    const path=`${CFG.audioDir}/${file}`;

    try{ await app.vault.createFolder(CFG.audioDir); }catch(e){ /* папка уже есть — норм */ }

    try{
        await app.vault.createBinary(path, await blob.arrayBuffer());
        lastSavedFile = file;   // запоминаем для хоткея вставки
        if(CFG.saveMode===2){
            insertLink(`![[${file}]]`);
            setStatus("Сохранено + ссылка вставлена","ok");
        } else {
            setStatus("Сохранено: "+file,"ok");
        }
        if(!expanded) setExpanded(true);
        loadRecent();
        if(finalTxt.trim()){ tBox.className="show"; tIns.classList.add("show"); }
    }catch(err){
        setStatus("Ошибка сохранения: "+err.message,"err");
    }
}

// ── Вставка ссылки / текста ───────────────────────────────────
function getEditor(){
    // Используем optional chaining — dv.current() может быть null в async-коллбэке
    const self=dv.current()?.file?.path;
    // Сначала пробуем последний активный лист
    const recent=app.workspace.getMostRecentLeaf();
    if(recent?.view?.editor && recent.view.file?.path!==self) return recent.view.editor;
    // Fallback: любой открытый markdown, не являющийся рекордером
    for(const l of app.workspace.getLeavesOfType("markdown"))
        if(l.view?.editor && l.view.file?.path!==self) return l.view.editor;
    return null;
}

function insertLink(text){
    const ed=getEditor();
    if(ed){ ed.replaceSelection(text+"\n"); }
    else  { navigator.clipboard?.writeText(text); new Notice("Ссылка скопирована: "+text,5000); }
}

tIns.addEventListener("click",()=>{
    const text=finalTxt.trim(); if(!text) return;
    const ed=getEditor();
    if(ed){ ed.replaceSelection(text+"\n"); setStatus("Текст вставлен","ok"); }
    else  { navigator.clipboard?.writeText(text); new Notice("Текст скопирован",4000); }
});

// ── Вставка по хоткею: текст транскрипции → иначе ссылка на последний файл ─
function doInsertByHotkey(){
    const text = finalTxt.trim();
    if(text){
        // Есть транскрипция — вставляем текст
        const ed=getEditor();
        if(ed){ ed.replaceSelection(text+"\n"); setStatus("Текст вставлен (хоткей)","ok"); }
        else  { navigator.clipboard?.writeText(text); new Notice("Текст скопирован",4000); }
    } else if(lastSavedFile){
        // Нет транскрипции, но есть сохранённый файл — вставляем ссылку
        insertLink(`![[${lastSavedFile}]]`);
        setStatus("Ссылка вставлена: "+lastSavedFile,"ok");
    } else {
        new Notice("Нечего вставлять — сначала сделайте запись",4000);
    }
}

async function loadRecent(){
    listEl.innerHTML="";
    const folder=app.vault.getAbstractFileByPath(CFG.audioDir);
    const files=(folder?.children||[])
        .filter(f=>["webm","mp3","m4a"].includes(f.extension))
        .sort((a,b)=>(b.stat?.mtime||0)-(a.stat?.mtime||0))
        .slice(0,6);

    if(!files.length){
        const em=document.createElement("div");
        em.className="vr-empty"; em.textContent="Записей нет";
        listEl.appendChild(em); return;
    }

    for(const f of files){
        const row=document.createElement("div"); row.className="vr-row";

        const nm=document.createElement("div"); nm.className="vr-rname";
        nm.textContent=f.name.replace(/\.(webm|mp3|m4a)$/,""); nm.title=f.name;

        const ins=document.createElement("button"); ins.className="vr-rb";
        ins.textContent="📎"; ins.title="Вставить ссылку в заметку";
        ins.addEventListener("click",()=>insertLink(`![[${f.name}]]`));

        let aud=null;
        const play=document.createElement("button"); play.className="vr-rb";
        play.textContent="▶"; play.title="Воспроизвести";
        play.addEventListener("click",()=>{
            if(aud && !aud.paused){ aud.pause(); play.textContent="▶"; return; }
            if(aud){ aud.currentTime=0; }
            else   { aud=new Audio(app.vault.getResourcePath(f)); aud.onended=()=>{ play.textContent="▶"; }; }
            aud.play(); play.textContent="⏸";
        });

        const del=document.createElement("button"); del.className="vr-rb vr-del";
        del.textContent="×"; del.title="Удалить";
        del.addEventListener("click", async ()=>{
            if(aud && !aud.paused){ aud.pause(); }
            try{
                await app.vault.trash(f, true); // в системную корзину
                row.remove();
                if(!listEl.querySelector(".vr-row")) loadRecent();
            }catch(e){ new Notice("Не удалось удалить: "+e.message, 4000); }
        });

        row.append(nm,ins,play,del);
        listEl.appendChild(row);
    }
}
