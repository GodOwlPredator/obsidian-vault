const fileLocks = {};
function withLock(p, fn) {
  const prev = fileLocks[p] || Promise.resolve();
  const next = prev.catch(() => {}).then(fn);
  fileLocks[p] = next;
  return next;
}

async function updateLine(p, idx, fn) {
  if (idx < 0) return Promise.resolve();
  return withLock(p, async () => {
    const f = app.vault.getAbstractFileByPath(p);
    if (!f) return;
    const c = await app.vault.read(f);
    const L = c.split("\n");
    if (idx >= L.length) return;
    const out = fn(L[idx]);
    if (out == null || out === L[idx]) return;
    L[idx] = out;
    await app.vault.modify(f, L.join("\n"));
  });
}

function setCheckbox(line, ch) {
  const re = /^(\s*(?:> ?)*-? ?\[)[ x\-X\/>](\]\s*)/;
  const m = line.match(re);
  if (m) return line.slice(0, m[1].length) + ch + line.slice(m[1].length + 1);
  return null;
}

const ROW_RE = /^(\s*[-*] )\[([ xX])\](.*)$/;
const WHEN_RE = /^(\d{4}-\d{2}-\d{2})\s+(\d{2}:\d{2})\s*[—-]\s*(.+)$/;

const pad = n => String(n).padStart(2, "0");
const fmtD = d => d.getFullYear() + "-" + pad(d.getMonth() + 1) + "-" + pad(d.getDate());
const fmtT = d => pad(d.getHours()) + ":" + pad(d.getMinutes());
const parseWhen = s => { const p = s.split(/[- :]/); return new Date(+p[0], +p[1] - 1, +p[2], +p[3], +p[4]); };

function splitText(t) {
  t = (t || "").trim();
  const m = t.match(/^(.+?)[.:]\s+(.+)$/);
  if (m && m[1].trim().length >= 2) return { title: m[1].trim(), desc: m[2].trim() };
  return { title: t, desc: "" };
}

let rows = [];
async function refreshRows() {
  const f = app.vault.getAbstractFileByPath(path);
  if (!f) return;
  const content = await app.vault.read(f);
  rows = content.split("\n").map((line, i) => ({ line, i })).filter(x => ROW_RE.test(x.line)).map(x => {
    const mm = ROW_RE.exec(x.line);
    const wm = WHEN_RE.exec(mm[3].trim());
    if (!wm) return null;
    return {
      i: x.i,
      done: mm[2] === "x" || mm[2] === "X",
      when: wm[1] + " " + wm[2],
      text: wm[3],
      title: splitText(wm[3]).title,
      desc: splitText(wm[3]).desc,
    };
  }).filter(Boolean);
  rows.sort((a, b) => String(a.when).localeCompare(String(b.when)));
}

// ── Время: колесо мыши (±1/±5, с wrap) + ручной ввод с клавиатуры ──
function clampNum(n, min, max) {
  n = parseInt(n, 10);
  if (isNaN(n)) return min;
  return Math.max(min, Math.min(max, n));
}
function mkTimeInp(val, min, max, step, wrap, label) {
  const inp = bar.createEl("input");
  inp.type = "text";
  inp.inputMode = "numeric";
  inp.maxLength = 2;
  inp.title = label;
  inp.value = pad(clampNum(val, min, max));
  inp.style.cssText = "width:44px;background:#121212;color:" + C.text + ";border:1px solid " + C.border + ";border-radius:6px;padding:7px 4px;font-size:13px;text-align:center;";
  inp.addEventListener("wheel", (e) => {
    e.preventDefault();
    const cur = clampNum(inp.value, min, max);
    let n = e.deltaY < 0 ? cur + step : cur - step;
    if (wrap) { if (n > max) n = min; else if (n < min) n = max; }
    else n = Math.max(min, Math.min(max, n));
    inp.value = pad(n);
  });
  inp.addEventListener("keydown", (e) => {
    if (e.key === "Enter") { e.preventDefault(); doAdd(); return; }
    if (/^[0-9]$/.test(e.key) || e.key === "Backspace" || e.key === "Delete" || e.key === "Tab"
        || e.key === "ArrowLeft" || e.key === "ArrowRight" || e.ctrlKey || e.metaKey) return;
    if (e.key === "ArrowUp") {
      e.preventDefault();
      const cur = clampNum(inp.value, min, max);
      inp.value = pad(wrap ? (cur + step > max ? min : cur + step) : Math.min(max, cur + step));
      return;
    }
    if (e.key === "ArrowDown") {
      e.preventDefault();
      const cur = clampNum(inp.value, min, max);
      inp.value = pad(wrap ? (cur - step < min ? max : cur - step) : Math.max(min, cur - step));
      return;
    }
    e.preventDefault();
  });
  inp.addEventListener("blur", () => { inp.value = pad(clampNum(inp.value, min, max)); });
  return inp;
}

function diffText(whenStr) {
  const t = parseWhen(whenStr);
  let s = Math.round((t - Date.now()) / 1000);
  const neg = s < 0; s = Math.abs(s);
  const d = Math.floor(s / 86400), h = Math.floor((s % 86400) / 3600), m = Math.floor((s % 3600) / 60);
  let txt = d > 0 ? d + " дн " : "";
  if (h > 0 || d > 0) txt += h + " ч ";
  txt += m + " мин";
  return (neg ? "Просрочено на " : "Через ") + txt;
}

async function del(row) {
  await withLock(path, async () => {
    const f = app.vault.getAbstractFileByPath(path);
    let c = await app.vault.read(f);
    const L = c.split("\n");
    if (row.i < L.length) {
      L.splice(row.i, 1);
      await app.vault.modify(f, L.join("\n"));
    }
  });
  await refreshRows();
  render();
}
