const fileLocks = {};
const pad = n => String(n).padStart(2, "0");
const fmtD = d => d.getFullYear() + "-" + pad(d.getMonth() + 1) + "-" + pad(d.getDate());
const fmtT = d => pad(d.getHours()) + ":" + pad(d.getMinutes());
const ROW_RE = /^\s*[-*] \[([ xX])\]\s+(\d{4}-\d{2}-\d{2})\s+(\d{2}:\d{2})\s*[—-]\s*(.+)$/;

function splitText(t) {
  t = (t || "").trim();
  const m = t.match(/^(.+?)[.:]\s+(.+)$/);
  if (m && m[1].trim().length >= 2) return { title: m[1].trim(), desc: m[2].trim() };
  return { title: t, desc: "" };
}

function withLock(p, fn) {
  const prev = fileLocks[p] || Promise.resolve();
  const next = prev.catch(() => {}).then(fn);
  fileLocks[p] = next;
  return next;
}

async function readList() {
  const f = app.vault.getAbstractFileByPath(LIST_PATH);
  if (!f) return [];
  const c = await app.vault.read(f);
  return c.split("\n").map((line, i) => ({ line, i })).filter(x => ROW_RE.test(x.line)).map(x => {
    const m = ROW_RE.exec(x.line);
    return {
      i: x.i,
      done: m[1] === "x" || m[1] === "X",
      when: m[2] + " " + m[3],
      text: m[4],
      title: splitText(m[4]).title,
      desc: splitText(m[4]).desc,
    };
  }).sort((a, b) => String(a.when).localeCompare(String(b.when)));
}

const parseWhen = s => { const p = s.split(/[- :]/); return new Date(+p[0], +p[1] - 1, +p[2], +p[3], +p[4]); };

function diffText(whenStr) {
  let s = Math.round((parseWhen(whenStr) - Date.now()) / 1000);
  const neg = s < 0; s = Math.abs(s);
  const d = Math.floor(s / 86400), h = Math.floor((s % 86400) / 3600), m = Math.floor((s % 3600) / 60);
  let txt = d > 0 ? d + " дн " : "";
  if (h > 0 || d > 0) txt += h + " ч ";
  txt += m + " мин";
  return (neg ? "🔔 просрочено " : "через ") + txt;
}

function bindTimeInp(inp, min, max, step, wrap, onEnter) {
  const clamp = n => {
    n = parseInt(n, 10);
    if (isNaN(n)) return min;
    return Math.max(min, Math.min(max, n));
  };
  const norm = () => { inp.value = pad(clamp(inp.value)); };
  inp.value = pad(clamp(inp.value));
  inp.addEventListener("wheel", (e) => {
    e.preventDefault();
    let n = clamp(inp.value);
    n = e.deltaY < 0 ? n + step : n - step;
    if (wrap) { if (n > max) n = min; else if (n < min) n = max; }
    else n = Math.max(min, Math.min(max, n));
    inp.value = pad(n);
  });
  inp.addEventListener("keydown", (e) => {
    if (e.key === "Enter") { e.preventDefault(); if (onEnter) onEnter(); return; }
    if (/^[0-9]$/.test(e.key) || e.key === "Backspace" || e.key === "Delete" || e.key === "Tab"
        || e.key === "ArrowLeft" || e.key === "ArrowRight" || e.ctrlKey || e.metaKey) return;
    if (e.key === "ArrowUp") {
      e.preventDefault();
      let n = clamp(inp.value) + step;
      if (wrap && n > max) n = min;
      inp.value = pad(Math.min(max, n));
      return;
    }
    if (e.key === "ArrowDown") {
      e.preventDefault();
      let n = clamp(inp.value) - step;
      if (wrap && n < min) n = max;
      inp.value = pad(Math.max(min, n));
      return;
    }
    e.preventDefault();
  });
  inp.addEventListener("blur", norm);
}

const HK_DEFAULTS = { hotkeyToggle: null, hotkeySnooze: null, hotkeyOpen: null };
function loadHk() { try { return { ...HK_DEFAULTS, ...JSON.parse(localStorage.getItem(HK_CFG_KEY) || "{}") }; } catch (e) { return { ...HK_DEFAULTS }; } }
function saveHk(c) { try { localStorage.setItem(HK_CFG_KEY, JSON.stringify(c)); } catch (e) {} }
let CFG = loadHk();

function fmtHk(hk) {
  if (!hk || (!hk.key && !hk.code)) return "";
  const p = [];
  if (hk.ctrl)  p.push("Ctrl");
  if (hk.alt)   p.push("Alt");
  if (hk.shift) p.push("Shift");
  let k;
  if (hk.code) {
    if     (hk.code.startsWith("Key"))    k = hk.code.slice(3);
    else if (hk.code.startsWith("Digit"))  k = hk.code.slice(5);
    else if (hk.code.startsWith("Numpad")) k = "Num" + hk.code.slice(6);
    else                                   k = hk.code;
  } else {
    k = hk.key === " " ? "Space" : hk.key.length === 1 ? hk.key.toUpperCase() : hk.key;
  }
  p.push(k);
  return p.join(" + ");
}
function matchHk(e, hk) {
  if (!hk || (!hk.key && !hk.code)) return false;
  const keyMatch = hk.code ? e.code === hk.code : e.key.toLowerCase() === hk.key;
  return keyMatch && e.ctrlKey === !!hk.ctrl && e.shiftKey === !!hk.shift && e.altKey === !!hk.alt;
}

function syncHkFields() {
  const ids = { "ocrm-hk1": "hotkeyToggle", "ocrm-hk2": "hotkeySnooze", "ocrm-hk3": "hotkeyOpen" };
  for (const id in ids) {
    const inp = g(id);
    inp.value = fmtHk(CFG[ids[id]]);
    inp.placeholder = inp.value ? "" : "Нажми…";
  }
}

function bindHk(inp, cfgKey) {
  inp.addEventListener("click", () => {
    inp.value = ""; inp.placeholder = "Нажми комбинацию…"; inp.classList.add("cap");
    inp.focus();
    const handler = (e) => {
      if (["Shift", "Control", "Alt", "Meta"].includes(e.key)) return;
      e.preventDefault(); e.stopPropagation();
      if (e.key === "Escape") {
        inp.classList.remove("cap"); syncHkFields();
        document.removeEventListener("keydown", handler, true);
        return;
      }
      CFG[cfgKey] = { key: e.key.toLowerCase(), code: e.code, ctrl: e.ctrlKey, shift: e.shiftKey, alt: e.altKey };
      saveHk(CFG);
      inp.value = fmtHk(CFG[cfgKey]); inp.placeholder = ""; inp.classList.remove("cap");
      document.removeEventListener("keydown", handler, true);
    };
    document.addEventListener("keydown", handler, true);
  });
}

bindHk(g("ocrm-hk3"), "hotkeyOpen");
