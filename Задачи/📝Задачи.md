function detectPriority(text, tags) {
  const t = (text || "").toLowerCase();
  const tg = (tags || []).map(x => x.replace(/^#/, "").toLowerCase());
  const has = (...ws) => ws.some(w => tg.some(x => x.includes(w)) || t.includes(w));
  if (has("критич", "срочн", "urgent")) return "критично";
  if (has("высок", "важн", "high"))     return "высокий";
  if (has("низк", "low"))               return "низкий";
  if (has("когда", "someday", "maybe")) return "когда-нибудь";
  return "средний";
}

function detectStatus(ch) {
  if (ch === "x" || ch === "X") return "готово";
  if (ch === "/" || ch === "-" || ch === ">") return "в работе";
  return "todo";
}

function cleanText(text) {
  return (text || "")
    .replace(/\[\[([^\]|]+)(\|[^\]]+)?\]\]/g, "$1")
    .replace(/#\S+/g, "")
    .replace(/\s{2,}/g, " ")
    .trim();
}

function folderOf(path) {
  const p = (path || "").split("/");
  return p.length > 1 ? p[0] : "";
}

function srcLabel(t) {
  const m = t.fileName.match(/^(\d{2})-([а-яё]+)-(\d{4})$/);
  if (m && MONTHS[m[2]]) return "📅 " + m[1] + "." + MONTHS[m[2]];
  return t.fileName;
}

// ── Сериализация записей в файл (без потери правок при быстрых

const fileLocks = {};
function withLock(path, fn) {
  const prev = fileLocks[path] || Promise.resolve();
  const next = prev.catch(() => {}).then(fn);
  fileLocks[path] = next;
  return next;
}

// Трансформация исходной строки ФАЙЛА (с префиксом "- [ ] ")
async function updateLine(path, lineIdx, fn) {
  if (lineIdx < 0) return Promise.resolve();
  return withLock(path, async () => {
    const file = app.vault.getAbstractFileByPath(path);
    if (!file) return;
    const content = await app.vault.read(file);
    const lines = content.split("\n");
    if (lineIdx >= lines.length) return;
    const out = fn(lines[lineIdx]);
    if (out == null || out === lines[lineIdx]) return;
    lines[lineIdx] = out;
    await app.vault.modify(file, lines.join("\n"));
  });
}

function setCheckbox(line, newCh) {
  const re = /^(\s*(?:> ?)*-? ?\[)[ x\->X\/](\]\s*)/;
  const m = line.match(re);
  if (m) return line.slice(0, m[1].length) + newCh + line.slice(m[1].length + 1);
  const m2 = line.match(/^(\s*-)\s+/);
  if (m2) return line.slice(0, m2[0].length) + "[" + newCh + "] " + line.slice(m2[0].length);
  return null;
}

async function addTask() {
  const text = (addInp.value || "").trim();
  if (!text) { new Notice("Напиши текст задачи"); return; }
  const prio = prioSel.value || "средний";
  const path = dv.current().file.path;
  const file = app.vault.getAbstractFileByPath(path);
  if (!file) { new Notice("Не найден файл списка"); return; }
  const line = "- [ ] " + cleanText(text) + " " + (PRIO_TAG[prio] || "#средний");
  let lineIdx = -1;
  await withLock(path, async () => {
    let content = await app.vault.read(file);
    if (!content.endsWith("\n")) content += "\n";
    const lines = content.split("\n");          // последний элемент — пустой (за \n)
    lineIdx = lines.length - 1;                  // индекс строки, которую дописываем
    lines[lines.length - 1] = line;
    await app.vault.modify(file, lines.join("\n") + "\n");
  });
  if (lineIdx < 0) return;
  allTasks.push({
    text: cleanText(text), rawText: cleanText(text),
    tags: [PRIO_TAG[prio] || "#средний"], ch: " ",
    status: "todo", priority: prio, path, line: lineIdx,
    folder: folderOf(path), fileName: path.split("/").pop().replace(/\.md$/i, ""),
  });
  state.draft = "";
  addInp.value = "";
  saveState();
  render();
  flashKey(keyOf({ path, line: lineIdx }));
}

function matches(t) {
  if (state.folder !== null && t.folder !== state.folder) return false;
  const q = state.q.trim().toLowerCase();
  if (!q) return true;
  return (t.text + " " + t.fileName + " " + t.folder + " " + t.tags.join(" ")).toLowerCase().includes(q);
}

function groupLabel(folder) {

}

function openFile(path) {
  const f = app.vault.getAbstractFileByPath(path);
  if (f) app.workspace.getLeaf(false).openFile(f);
  else   app.workspace.openLinkText(path, "", false);
}

function flashKey(key) {
  let sel = "";
  try { sel = '[data-key="' + CSS.escape(key) + '"]'; } catch (_) { return; }
  setTimeout(() => {
    let el = null;
    try { el = document.querySelector(sel); } catch (_) {}
    if (!el) return;
    el.classList.add("vt4-flash");
    setTimeout(() => el.classList.remove("vt4-flash"), 720);
    try { el.scrollIntoView({ block: "nearest" }); } catch (_) {}
  }, 420);
}

async function toggleDone(task, ref) {
  const next = task.status === "готово" ? "todo" : "готово";
  const newCh = next === "готово" ? "x" : " ";
  task.status = next;
  task.ch = newCh;
  if (ref) {
    ref.cbox.classList.toggle("on", next === "готово");
    ref.cbox.textContent = next === "готово" ? "✓" : "";
    ref.cbox.title = next === "готово" ? "Отменить выполнение" : "Отметить выполненным";
    ref.a.style.color = next === "готово" ? C.dim : C.text;
    ref.a.style.textDecoration = next === "готово" ? "line-through" : "none";
  } else {
    render();
  }
  await updateLine(task.path, task.line, (line) => setCheckbox(line, newCh));
  flashKey(keyOf(task));
}

async function cyclePriority(task, ref) {
  const next = PRIO_ORDER[(PRIO_ORDER.indexOf(task.priority) + 1) % PRIO_ORDER.length];
  task.priority = next;
  task.rawText = task.rawText.replace(PRIO_RE, "").replace(/\s{2,}/g, " ").trimEnd() + " " + PRIO_TAG[next];
  if (ref) {
    ref.dot.style.background = PRIO_DOT[next];
    ref.dot.title = "Приоритет: " + next + " (клик — сменить)";
  } else {
    render();
  }
  await updateLine(task.path, task.line, (line) =>
    line.replace(PRIO_RE, "").replace(/\s{2,}/g, " ").trimEnd() + " " + PRIO_TAG[next]);
  flashKey(keyOf(task));
}
