function buildRows(iso, name) {
    const base = progExercises(name);
    let stored = [];
    if (sessionExists(iso, name)) {
        try { stored = objArr(dv.page(sessionPath(iso, name)).exercises) || []; } catch (e) { stored = []; }
    }
    // Факты привязываем к упражнению по имени (иначе правка программы сдвигает результаты)
    const used = new Array(stored.length).fill(false);
    const rows = base.map((e, i) => {
        let idx = stored.findIndex((s, j) => !used[j] && String(s.name || "") === e.name);
        if (idx < 0 && i < stored.length && !used[i] && !String(stored[i].name || "").trim()) idx = i;
        if (idx >= 0) used[idx] = true;
        const r = {
            name: e.name, sets: e.sets, reps: e.reps, weight: e.weight, rest: e.rest,
            note: e.note, done: false, fact_weight: 0, fact_reps: 0,
        };
        if (idx >= 0) {
            const s = plain(stored[idx]);
            if (s.sets   != null && s.sets   !== "") r.sets = Number(s.sets) || r.sets;
            if (s.reps   != null && s.reps   !== "") r.reps = Number(s.reps) || r.reps;
            if (s.weight != null && s.weight !== "") r.weight = Number(s.weight) || r.weight;
            if (s.rest   != null && s.rest   !== "") r.rest = Number(s.rest) || r.rest;
            r.note = r.note || String(s.note || "");
            r.done = !!s.done;
            r.fact_weight = Number(s.fact_weight) || 0;
            r.fact_reps   = Number(s.fact_reps)   || 0;
        }
        return r;
    });
    // Добавленные вручную упражнения (нет их в программе)
    for (let j = 0; j < stored.length; j++) {
        if (used[j]) continue;
        const s = plain(stored[j]);
        rows.push({
            name: String(s.name || ""), sets: Number(s.sets) || 3, reps: Number(s.reps) || 10,
            weight: Number(s.weight) || 0, rest: Number(s.rest) || 60, note: String(s.note || ""),
            done: !!s.done, fact_weight: Number(s.fact_weight) || 0, fact_reps: Number(s.fact_reps) || 0,
        });
    }
    return rows;
}

async function ensureSessionFile(iso, name) {
    const path = sessionPath(iso, name);
    let f = app.vault.getAbstractFileByPath(path);
    if (f) return f;
    try { await app.vault.createFolder(TRAININGS_DIR); } catch (e) {}
    const m = progMeta(name);
    const icon = m ? m.icon : "🏋️";
    f = await app.vault.create(path, `# ${icon} ${name} — ${dispOf(iso)}\n\n[[🏋️ Спорт/📊 Дашборд|📊 Дашборд]]\n`);
    const rows = progExercises(name).map(e => ({
        name: e.name, sets: e.sets, reps: e.reps, weight: e.weight, rest: e.rest,
        done: false, fact_weight: 0, fact_reps: 0,
    }));
    const m2 = progMeta(name);
    await app.fileManager.processFrontMatter(f, fm => {
        fm.log = "спорт";
        fm.date = iso;
        fm.program = name;
        fm.type = m2 ? m2.type : name;
        fm["иконка"] = icon;
        fm.duration = 0; fm.intensity = 0; fm.energy = 0; fm.mood = 0;
        fm["готово"] = false;
        fm.exercises = rows;
    });
    return app.vault.getAbstractFileByPath(path) || f;
}

async function persistSession(iso, name, rows) {
    const f = await ensureSessionFile(iso, name);
    await app.fileManager.processFrontMatter(f, fm => {
        fm.exercises = rows.map(r => ({
            name: r.name, sets: Number(r.sets) || 0, reps: Number(r.reps) || 0,
            weight: Number(r.weight) || 0, rest: Number(r.rest) || 0,
            done: !!r.done,
            fact_weight: Number(r.fact_weight) || 0,
            fact_reps:   Number(r.fact_reps)   || 0,
        }));
        fm["готово"] = rows.length > 0 && rows.every(r => r.done);
    });
    return f;
}

/* ── Модалка ─────────────────────────────────────────────────────────── */
function buildModal(title) {
    const prev = document.getElementById("sd-overlay");
    if (prev) prev.remove();
    const ov = h("div", "sd-overlay"); ov.id = "sd-overlay";
    const box = h("div", "sd-modal"); ov.appendChild(box);
    const hd = h("div", "sd-modal-hd"); box.appendChild(hd);
    hd.appendChild(h("div", "sd-modal-title", title));
    const x = h("button", "sd-modal-x", "✕"); x.type = "button"; hd.appendChild(x);
    const bodyEl = h("div", "sd-modal-body"); box.appendChild(bodyEl);
    function close() { ov.remove(); document.removeEventListener("keydown", esc); }
    function esc(e) { if (e.key === "Escape") close(); }
    document.addEventListener("keydown", esc);
    ov.addEventListener("click", e => { if (e.target === ov) close(); });
    x.onclick = close;
    document.body.appendChild(ov);
    return { ov, body: bodyEl, close };
}

function pickProgram(title, onPick, opts) {
    opts = opts || {};
    const m = buildModal(title);
    const names = [];

    for (const n of names) {

function renderSessionCard(parent, iso, name, opts) {
    opts = opts || {};
    const meta = progMeta(name);
    const icon = meta ? meta.icon : "🏋️";
    const rows = buildRows(iso, name);

    const card = h("div", "sd-today" + (opts.rest ? " rest" : ""));
    parent.appendChild(card);

    const head = h("div", "sd-s-head"); card.appendChild(head);
    const left = h("div"); head.appendChild(left);
    const ttl = h("div", "sd-s-title");
    ttl.appendChild(h("span", null, icon));
    ttl.appendChild(h("span", null, opts.title || name));
    left.appendChild(ttl);
    const sub = h("div", "sd-s-sub"); left.appendChild(sub);
    const mins = meta ? meta.mins : 0;
    const doneCount = () => rows.filter(r => r.done).length;
    function updSub() {
        const parts = [];
        if (mins) parts.push("~" + mins + " мин");
        parts.push(rows.length + " упражн.");
        sub.textContent = parts.join(" · ");
    }
    updSub();

    const right = h("div", "sd-s-right"); head.appendChild(right);
    const progEl = h("span", "sd-s-progress"); right.appendChild(progEl);
    function updProg() {
        const d = doneCount();
        progEl.textContent = d + "/" + rows.length;
        progEl.className = "sd-s-progress" + (rows.length && d === rows.length ? " ok" : "");
    }
    const openBtn = h("button", "sd-mini-btn", "Открыть"); openBtn.type = "button";
    openBtn.onclick = async () => { await ensureSessionFile(iso, name); openSession(iso, name); };
    right.appendChild(openBtn);
    updProg();

    const list = h("div", "sd-ex"); card.appendChild(list);

    function makeRow(r, i) {
        const rowEl = h("div", "sd-ex-row" + (r.done ? " done" : ""));
        const chk = h("button", "sd-chk" + (r.done ? " on" : ""), "✓");
        chk.type = "button";
        chk.setAttribute("aria-label", r.name + " — выполнено");
        chk.setAttribute("aria-pressed", r.done ? "true" : "false");
        chk.onclick = async () => {
            rows[i].done = !rows[i].done;
            chk.className = "sd-chk" + (rows[i].done ? " on" : "");
            chk.setAttribute("aria-pressed", rows[i].done ? "true" : "false");
            rowEl.className = "sd-ex-row" + (rows[i].done ? " done" : "");
            updProg();
            try { await persistSession(iso, name, rows); } catch (e) { new Notice("⚠ " + e.message); }
        };
        rowEl.appendChild(chk);

        const nm = h("div", "sd-ex-name", r.name);
        if (r.note) nm.appendChild(h("span", "sd-ex-note", r.note));
        rowEl.appendChild(nm);

        const plan = h("div", "sd-ex-plan");
        plan.textContent = `${r.sets}×${r.reps}` + (r.weight ? ` · ${r.weight} кг` : "");
        rowEl.appendChild(plan);

        const fact = h("div", "sd-ex-fact");
        const wi = h("input", "sd-in"); wi.type = "text"; wi.inputMode = "decimal";
        wi.placeholder = "—"; wi.value = r.fact_weight ? String(r.fact_weight) : "";
        wi.title = "Фактический вес, кг";
        wi.setAttribute("aria-label", "Фактический вес, кг");
        const xi = h("span", "sd-u", "кг ×");
        const ri = h("input", "sd-in"); ri.type = "text"; ri.inputMode = "numeric";
        ri.placeholder = "—"; ri.value = r.fact_reps ? String(r.fact_reps) : "";
        ri.title = "Фактические повторы";
        ri.setAttribute("aria-label", "Фактические повторы");
        const xu = h("span", "sd-u", "повт");
        async function saveFact() {
            rows[i].fact_weight = parseFloat(String(wi.value).replace(",", ".")) || 0;
            rows[i].fact_reps = parseInt(String(ri.value).replace(/[^\d]/g, ""), 10) || 0;
            try { await persistSession(iso, name, rows); } catch (e) { new Notice("⚠ " + e.message); }
        }
        wi.onchange = saveFact; ri.onchange = saveFact;
        fact.appendChild(wi); fact.appendChild(xi); fact.appendChild(ri); fact.appendChild(xu);
        rowEl.appendChild(fact);

        return rowEl;
    }
    function renderRows() {
        list.innerHTML = "";
        if (rows.length === 0) {
            const em = h("div", "sd-empty", "Упражнений нет.");
            em.style.cssText = "color:" + C.muted + ";font-size:12px;padding:8px 0;";
            list.appendChild(em);
        }
        rows.forEach((r, i) => list.appendChild(makeRow(r, i)));
    }
    function appendRow(r) {
        const em = list.querySelector(".sd-empty");
        if (em) em.remove();
        rows.push(r);
        list.appendChild(makeRow(r, rows.length - 1));
        updSub(); updProg();
    }
    renderRows();

    const addBtn = h("button", "sd-add", "＋ упражнение");
    addBtn.type = "button";
    addBtn.onclick = () => {
        const m = buildModal("Добавить упражнение");
        const mk = (lbl, ph) => {
            m.body.appendChild(h("div", "sd-field-lbl", lbl));
            const inp = h("input", "sd-field-inp"); inp.type = "text"; inp.placeholder = ph || "";

            m.close();
            appendRow({
                name: nmv, sets: parseInt(sInp.value, 10) || 3, reps: parseInt(rInp.value, 10) || 10,
                weight: parseFloat(String(wInp.value).replace(",", ".")) || 0, rest: 60,
                done: false, fact_weight: 0, fact_reps: 0,
            });
            try { await persistSession(iso, name, rows); } catch (e) { new Notice("⚠ " + e.message); }
        };
        // Enter в любом поле = «Добавить»
        for (const inp of [nInp, sInp, rInp, wInp]) {
            inp.addEventListener("keydown", e => {
                if (e.key === "Enter") { e.preventDefault(); go.click(); }
            });
        }
        m.body.appendChild(go);
        setTimeout(() => nInp.focus(), 50);
    };
    card.appendChild(addBtn);

    /* ── Как прошло: длительность / интенсивность / энергия / самоощущение ── */
    const well = sessionWell(iso, name);
    const wellBox = h("div", "sd-well");
    wellBox.appendChild(h("div", "sd-well-title", "Как прошло?"));
    let wellTimer = null;
    function saveWell() {
        clearTimeout(wellTimer);
        wellTimer = setTimeout(() => {
            persistWell(iso, name, well).catch(e => new Notice("⚠ " + e.message));
        }, 400);
    }

        const row = h("div", "sd-well-row");

            c.type = "button";
            c.title = lbl + ": " + v + " из " + max;
            c.setAttribute("aria-label", lbl + " " + v + " из " + max);
            c.setAttribute("aria-pressed", well[key] === v ? "true" : "false");
            c.onclick = () => {
                well[key] = well[key] === v ? 0 : v;   // повторный клик — сброс
                for (const sib of box.children) {
                    const on = Number(sib.textContent) === well[key];
                    sib.className = "sd-chip" + (on ? " on" : "");
                    sib.setAttribute("aria-pressed", on ? "true" : "false");
                }
                saveWell();
            };
            box.appendChild(c);
        }
        row.appendChild(box);
        wellBox.appendChild(row);
    }
    // ⏱ минуты — числовое поле
    {
        const row = h("div", "sd-well-row");
        row.appendChild(h("div", "sd-well-lbl", "⏱ Длительность"));
        const inp = h("input", "sd-in-min"); inp.type = "text"; inp.inputMode = "numeric";
        inp.placeholder = "мин"; inp.value = well.duration ? String(well.duration) : "";
        inp.title = "Сколько минут длилась тренировка";
        inp.setAttribute("aria-label", "Длительность в минутах");
        inp.onchange = () => { well.duration = parseInt(String(inp.value).replace(/[^\d]/g, ""), 10) || 0; saveWell(); };
        row.appendChild(inp);
        row.appendChild(h("div", "sd-well-hint", "минут"));
        wellBox.appendChild(row);
    }
    chipRow("🔥", "Интенсивность", 10, "intensity");
    chipRow("⚡", "Энергия", 5, "energy");

    return card;

    todayWrap.innerHTML = "";

        const c = renderSessionCard(todayWrap, TODAY_ISO, pl, { title: pl });

    for (const raw of names) {
        const nm = String(raw || "").trim();

        const c = renderSessionCard(todayWrap, TODAY_ISO, nm, { title: nm + " · сегодня" });
        c.dataset.sess = sessKey(TODAY_ISO, nm);
        shown.add(norm(nm));
    }
    if (shown.size === 0) renderRestCard(todayWrap, TODAY_ISO);
    todayShown = shown;
}

const PROFILE_PATH = DIR + "/Профиль.md";
let profCache = null;   // после сохранения dv может узнать о файле не сразу — доверяем кэшу

function readProfile() {
    if (profCache) return profCache;
    let p = null;
    try { p = dv.page(PROFILE_PATH); } catch (e) { p = null; }
    const hist = p ? objArr(p["вес_история"]) : [];
    return {
        height: p ? Number(p["рост"]) || 0 : 0,
        weight: p ? Number(p["вес"]) || 0 : 0,
        hist: hist
            .map(x => ({ d: pDate(x.дата || x.date), w: Number(x.вес || x.weight) || 0 }))
            .filter(x => x.d && x.w > 0)
            .sort((a, b) => (a.d < b.d ? -1 : a.d > b.d ? 1 : 0)),
    };
}

async function saveProfile(height, weight) {
    const prev = readProfile();
    let f = app.vault.getAbstractFileByPath(PROFILE_PATH);
    if (!f) f = await app.vault.create(PROFILE_PATH, "# 🧍 Профиль\n\n[[🏋️ Спорт/📊 Дашборд|📊 Дашборд]]\n");
    let hist = prev.hist.slice();
    if (weight > 0) {
        hist = hist.filter(x => x.d !== TODAY_ISO);   // одна запись в день
        hist.push({ d: TODAY_ISO, w: weight });
        hist.sort((a, b) => (a.d < b.d ? -1 : a.d > b.d ? 1 : 0));
    }
    await app.fileManager.processFrontMatter(f, fm => {
        fm.log = "профиль";
