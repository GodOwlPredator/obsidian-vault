function toDate(d) {
    if (!d) return null;
    if (typeof d === "object" && typeof d.toFormat === "function") return d;

    const eu = s.match(/^(\d{2})\.(\d{2})\.(\d{4})$/);

// ── Создание файла из шаблона / пустого ─────────────────────
async function doCreate(tplFile, name, type) {
    const date    = todayISO();
    const dateDis = todayDisplay();
    const fileName = `${name} ${todayFilename()}`;
    const destPath = `${CFG.trainingsDir}/${fileName}.md`;

    if (app.vault.getAbstractFileByPath(destPath)) {
        new Notice(`Файл уже существует: ${fileName}`, 4000);
        app.workspace.openLinkText(destPath, "", false);
        return true;
    }

    let content;
    if (tplFile) {
        content = await app.vault.read(tplFile);
        content = content.replace(/TPL_DATE_ISO/g, date).replace(/TPL_DATE/g, dateDis);
        // Без log/date/type созданная тренировка молча не попадёт в статистику
        const t2 = type || name;
        if (!content.startsWith("---")) {
            content = `---\nlog: спорт\ndate: ${date}\ntype: ${t2}\n---\n\n` + content;
        } else {
            if (!/^log:/m.test(content))  content = content.replace(/^---\n/, `---\nlog: спорт\n`);
            if (!/^date:/m.test(content)) content = content.replace(/^---\n/, `---\ndate: ${date}\n`);
            if (!/^type:/m.test(content)) content = content.replace(/^---\n/, `---\ntype: ${t2}\n`);
        }
    } else {
        const t = type || name;
        content = [
            "---", "log: спорт", `date: ${date}`, `type: ${t}`,
            "duration: 0", "intensity: 0", "energy: 0", "mood: 0", "notes:", "---",
            "", `# 🏋️ Тренировка — ${dateDis}`, "",
            "## 📋 Упражнения", "",
            "| Упражнение | Подходы | Повторения | Вес (кг) | Заметки |",
            "| ---------- | ------- | ---------- | -------- | ------- |",
            "|            |         |            |          |         |",
            "", "---", "", "## 💬 Заметки", "", "", "",
            "[[🏋️ Спорт/📊 Дашборд|📊 Дашборд]]",
        ].join("\n");
    }

    try { await app.vault.createFolder(CFG.trainingsDir); } catch(e) {}
    try {
        await app.vault.create(destPath, content);
        new Notice(`✅ Создана: ${fileName}`, 3000);
        app.workspace.openLinkText(destPath, "", false);
        return true;
    } catch(e) {
        new Notice(`❌ Ошибка: ${e.message}`, 8000);
        return false;
    }
}

// ── Копирование существующей тренировки ─────────────────────
async function doCopy(srcPage) {
    const date    = todayISO();
    const dateDis = todayDisplay();

    // Убираем дату из имени файла если есть
    const baseName = srcPage.file.name
        .replace(/^\d{4}-\d{2}-\d{2}\s*/, "")
        .replace(/^(\d{2})\.(\d{2})\.(\d{4})\s*/, "")
        .replace(/\s*\d{2}-\d{2}-\d{4}$/, "");

    const fileName = `${baseName} ${todayFilename()}`;
    const destPath = `${CFG.trainingsDir}/${fileName}.md`;

    if (app.vault.getAbstractFileByPath(destPath)) {
        new Notice(`Файл уже существует: ${fileName}`, 4000);
        app.workspace.openLinkText(destPath, "", false);
        return true;
    }

    const srcFile = app.vault.getAbstractFileByPath(srcPage.file.path);
    if (!srcFile) { new Notice("Файл не найден", 4000); return false; }

    let content = await app.vault.read(srcFile);

    // Заменяем дату в frontmatter (любой формат после "date:")
    content = content
        .replace(/^(date:\s*).*$/m, `$1${date}`)
        .replace(/TPL_DATE_ISO/g, date)
        .replace(/TPL_DATE/g, dateDis);

    // Дата в заголовке вида "— 01.06.2026" или "— 2026-06-13"
    // [\d.\-]+ матчит и точечные (01.06.2026) и ISO-даты (2026-06-13)
    content = content.replace(
        /(^#\s+.+?[—–-]\s*)[\d.\-]+/m,
        `$1${dateDis}`
    );

    // Сбрасываем шкалы
    content = content
        .replace(/^(intensity:\s*)\d+/m, "$10")
        .replace(/^(energy:\s*)\d+/m,    "$10")
        .replace(/^(mood:\s*)\d+/m,      "$10")
        .replace(/^(duration:\s*)\d+/m,  "$10");

    try { await app.vault.createFolder(CFG.trainingsDir); } catch(e) {}
    try {
        await app.vault.create(destPath, content);
        new Notice(`✅ Скопирована: ${fileName}`, 3000);
        app.workspace.openLinkText(destPath, "", false);
        return true;
    } catch(e) {
        new Notice(`❌ Ошибка: ${e.message}`, 8000);
        return false;
    }
}

            resetBtn.onmouseleave = () => { resetBtn.style.color = "#555"; resetBtn.style.borderColor = "#333"; };
            resetBtn.onclick = () => {
                inpTpl.value = DEFAULTS.templatesDir;
                inpTrn.value = DEFAULTS.trainingsDir;
            };
            resetRow.appendChild(resetBtn);
            body.appendChild(resetRow);

            const saveBtn = document.createElement("button");

                if (!tDir || !wDir) {

                saveCfg(CFG);
                new Notice("✅ Настройки сохранены", 2500);

                // Переоткрываем на вкладке шаблонов чтобы пересканировать папку

            body.appendChild(saveBtn);
        }
    }

    // Инициализация табов и рендер
    setActiveTab(tab);
