const vault = app.vault;
const dictPath = "Учёба 🎓/Предметы/.en-dictionary.json";

function loadDict(){
  try {
    if(!vault.adapter.exists(dictPath)) return [];
    const raw = vault.adapter.read(dictPath);
    const data = JSON.parse(raw);
    return Array.isArray(data) ? data : [];
  } catch(e){ return []; }
}
function saveDict(arr){
  vault.adapter.write(dictPath, JSON.stringify(arr, null, 2));
}

function detectLang(s){
  if(/[а-яё]/i.test(s)) return "ru";
  if(/[a-z]/i.test(s)) return "en";
  return "en";
}

async function translateWithEnTr(text){
  const {exec} = require('child_process');
  return new Promise((resolve)=>{
    exec('en-translate --json "' + text.replace(/"/g,'\\"') + '"', {timeout:10000}, (err,stdout)=>{
      try {
        const j = JSON.parse(stdout||'{}');
        resolve(j);
      } catch(e){ resolve({result:'', phonetic:'', dir:detectLang(text)}); }
    });
  });
}

async function getPhonetic(text){
  const {exec} = require('child_process');
  return new Promise((resolve)=>{
    exec('en-translate --phonetic "' + text.replace(/"/g,'\\"') + '"', {timeout:8000}, (err,stdout)=>{
      resolve((stdout||'').trim());
    });
  });
}

function renderDict(){
  const list = document.getElementById('en-dict-list');
  if(!list) return;
  const search = (document.getElementById('en-dict-search').value||'').toLowerCase();
  const data = loadDict().filter(x=>{
    if(!search) return true;
    return (x.word||'').toLowerCase().includes(search) || (x.trans||'').toLowerCase().includes(search);
  });
  list.innerHTML = '';
  for(const item of data){
    const row = document.createElement('div');
    row.className = 'en-dict-row';
    row.innerHTML = '<div class="en-cell">'+(item.word||'')+'</div><div class="en-cell">'+(item.trans||'')+'</div><div class="en-cell">'+(item.phonetic||'')+'</div>';
    list.appendChild(row);
  }
}

function setup(){
  const word = document.getElementById('en-word');
  const trans = document.getElementById('en-trans');
  const transcr = document.getElementById('en-transcr');
  const status = document.getElementById('en-status');
  const btnAdd = document.getElementById('en-btn-add-dict');
  const btnStudy = document.getElementById('en-btn-study');
  const btnStudyDict = document.getElementById('en-btn-study-dict');
  const search = document.getElementById('en-dict-search');

  document.querySelectorAll('.en-tab-btn').forEach(btn=>{
    btn.onclick = ()=>{
      document.querySelectorAll('.en-tab-btn').forEach(b=>b.classList.remove('active'));
      document.querySelectorAll('.en-tab').forEach(t=>t.classList.remove('active'));
      btn.classList.add('active');
      document.getElementById('en-tab-'+btn.dataset.tab).classList.add('active');
      if(btn.dataset.tab==='dict') renderDict();
    };
  });

  let timer = null;
  word.oninput = ()=>{
    clearTimeout(timer);
    const v = word.value.trim();
    if(!v){ trans.value=''; transcr.value=''; status.textContent=''; return; }
    timer = setTimeout(async ()=>{
      status.textContent = 'Перевод...';
      const j = await translateWithEnTr(v);
      trans.value = j.result || '';
      const ph = await getPhonetic(v);
      transcr.value = ph || j.phonetic || '';
      status.textContent = '';
    }, 400);
  };

  btnAdd.onclick = ()=>{
    const w = word.value.trim(), t = trans.value.trim(), p = transcr.value.trim();
    if(!w){ status.textContent='Нужно ввести слово/фразу'; return; }
    if(!t){ status.textContent='Нет перевода'; return; }
    const dict = loadDict();
    if(dict.find(x=> (x.word||'').toLowerCase()===w.toLowerCase())){ status.textContent='Уже в словаре'; return; }
    dict.push({word:w, trans:t, phonetic:p, added:new Date().toISOString()});
    saveDict(dict);
    word.value=''; trans.value=''; transcr.value='';
    status.textContent='Добавлено в словарь';
    renderDict();
  };

  btnStudy.onclick = btnStudyDict.onclick = ()=>{
    try { app.openLinkText('obsidian://open?vault=Obsidian%20Vault&file=Учёба%20🎓%2FПредметы%2F%F0%9F%93%9A%20Мои%20слова.md'); } catch(e){}
    const {exec}=require('child_process');
    exec('anki &');
  };

  if(search){ search.oninput = renderDict; }
  renderDict();
}

if(typeof app !== 'undefined'){ app.workspace.onLayoutReady(setup); }
