const C = {
    accent: "#8f0000", accentLight: "#b30000", card: "#1e1e1e", border: "#2e2e2e",
    text: "#e0e0e0", muted: "#777", green: "#2d7a2d", yellow: "#6a6000", empty: "#2a2a2a",
};
const P     = dv.current();
const SUB   = String(P.name || P.file.name);
const EMO   = String(P.emoji || "");
const TODAY = dv.date("today");

function toDate(d) {
    if (!d) return null;
    if (typeof d === "object" && typeof d.toFormat === "function") return d;
    try { return dv.date(String(d)); } catch (e) { return null; }
}
function iso(d)  { const x = toDate(d); return x ? x.toFormat("yyyy-MM-dd") : String(d || ""); }
function fmt(d)  { const x = toDate(d); return x ? x.toFormat("dd.MM.yyyy") : String(d || ""); }
function safe(s) { return String(s || "").replace(/[\\/:*?"<>|]/g, "-").trim().slice(0, 40); }

async function setValue(path, key, val) {
    const tf = app.vault.getAbstractFileByPath(path);
    if (!tf) return;
    let content = await app.vault.read(tf);
    const re = new RegExp("^(" + key + ":\\s*).*$", "m");
    content = re.test(content)
        ? content.replace(re, "$1" + val)
        : content.replace(/^---\n/, "---\n" + key + ": " + val + "\n");
    await app.vault.modify(tf, content);
}

function uniquePath(folder, base) {
    let p = folder + base + ".md", i = 2;
    while (app.vault.getAbstractFileByPath(p)) p = folder + base + " (" + i + ").md", i++;
    return p;
}

async function createFrom(tpl, tokens, folder, base) {
    const t = app.vault.getAbstractFileByPath(tpl);
    if (!t) { new Notice("Нет шаблона: " + tpl); return; }
    let s = await app.vault.read(t);
    for (const k in tokens) s = s.split("@" + k + "@").join(tokens[k]);
    const path = uniquePath(folder, safe(base));
    await app.vault.create(path, s);
    app.workspace.openLinkText(path, "", false);
}

function badge(txt, bg) {
    const b = document.createElement("span");
    b.textContent = txt;
    b.style.cssText = "display:inline-block;padding:3px 10px;border-radius:5px;font-size:10px;font-weight:700;text-transform:uppercase;color:#fff;background:" + (bg || C.muted) + ";";
    return b;
}
