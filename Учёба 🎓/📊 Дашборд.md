const C = {
    accent: "#8f0000", accentLight: "#b30000", card: "#1e1e1e", border: "#2e2e2e",
    text: "#e0e0e0", muted: "#777", green: "#2d7a2d", yellow: "#6a6000", empty: "#2a2a2a",
};
const P     = dv.current();
const TODAY = dv.date("today");
const SUBJ  = dv.pages().where(s => s.type === "предмет" && s.name && !s.file.path.includes("/Templates/"));
const EMO_OF = {};
for (const s of SUBJ.array()) EMO_OF[String(s.name || s.file.name)] = String(s.emoji || "");
const TASKS = dv.pages().where(t => t.type === "задание" && t.task && !t.file.path.includes("/Templates/"));
const CONS  = dv.pages().where(c => c.type === "конспект" && !c.file.path.includes("/Templates/"));

function toDate(d) {
    if (!d) return null;
    if (typeof d === "object" && typeof d.toFormat === "function") return d;
    try { return dv.date(String(d)); } catch (e) { return null; }
}
function iso(d) { const x = toDate(d); return x ? x.toFormat("yyyy-MM-dd") : String(d || ""); }
function badge(txt, bg) {
    const b = document.createElement("span");
    b.textContent = txt;
    b.style.cssText = "display:inline-block;padding:3px 10px;border-radius:5px;font-size:10px;font-weight:700;text-transform:uppercase;color:#fff;background:" + (bg || C.muted) + ";";
    return b;
}
function chips(label, val, color) {
    const c = div("div");
    c.style.cssText = "background:" + C.card + ";border:1px solid " + C.border + ";border-radius:8px;padding:8px 14px;text-align:center;";
    c.createEl("div", { text: String(val) }).style.cssText = "font-size:20px;font-weight:700;color:" + (color || C.accent) + ";line-height:1.2;";
    c.createEl("div", { text: label }).style.cssText = "font-size:10px;color:" + C.muted + ";text-transform:uppercase;letter-spacing:0.05em;";
    return c;
}

const root = dv.el("div", "");
root.style.cssText = "font-family:sans-serif;color:" + C.text;
function div(cls) { const e = root.createEl("div"); if (cls) e.style.cssText = cls; return e; }

const head = div("background:" + C.card + ";border:1px solid " + C.border + ";border-left:4px solid " + C.accent + ";border-radius:10px;padding:14px 16px;margin-bottom:12px;");
head.createEl("h3", { text: String(P.специальность || "") }).style.cssText = "margin:0 0 6px;font-size:15px;color:" + C.text + ";line-height:1.35;";
if (P.цель) head.createEl("div", { text: "🎯 " + P.цель }).style.cssText = "font-size:13px;color:" + C.muted + ";margin-bottom:10px;";
const headMeta = head.createEl("div");
headMeta.style.cssText = "display:flex;gap:8px;flex-wrap:wrap;";
headMeta.appendChild(badge("курс " + (P.курс || "—"), C.accent));
headMeta.appendChild(badge("семестр " + (P.семестр || "—"), "#5a5a00"));
headMeta.appendChild(badge(SUBJ.length + " предметов", C.green));

const stats = div("display:flex;gap:12px;flex-wrap:wrap;margin-bottom:12px;");
const activeTasks = TASKS.where(t => t.status !== "готово").array();
const overdue = activeTasks.filter(t => { const d = toDate(t.due); return d && d < TODAY; });
const soon    = activeTasks.filter(t => { const d = toDate(t.due); return d && d >= TODAY && d < TODAY.plus({ days: 8 }); });
stats.appendChild(chips("ДЗ в работе", activeTasks.length, C.yellow));
stats.appendChild(chips("просрочено", overdue.length, C.accentLight));
stats.appendChild(chips("дедлайн до 7 дн", soon.length, C.accent));
stats.appendChild(chips("конспектов", CONS.length, C.text));

const nav = div("display:flex;gap:8px;flex-wrap:wrap;margin-bottom:14px;");
function navBtn(text, path, primary) {
    const b = nav.createEl("button", { text: text });
    b.style.cssText = "padding:7px 16px;border-radius:8px;border:1px solid " + (primary ? C.accent : C.border) + ";background:" + (primary ? C.accent : C.card) + ";color:" + (primary ? "#fff" : C.text) + ";font-size:13px;cursor:pointer;";
    b.addEventListener("click", () => { app.workspace.openLinkText(path, "", false); });
    return b;
}
const path_sched = "Учёба 🎓/📅 Расписание.md", path_tasks = "Учёба 🎓/Задания/✅ Трекер заданий.md";
navBtn("📅 Расписание", path_sched, true);
navBtn("🎯 ДЗ и дедлайны", path_tasks, false);

div("font-size:11px;font-weight:600;color:" + C.muted + ";text-transform:uppercase;letter-spacing:0.08em;margin:4px 0 8px;").textContent = "Предметы";

const grid = div("display:grid;grid-template-columns:repeat(auto-fill,minmax(220px,1fr));gap:8px;margin-bottom:14px;");
for (const s of SUBJ.sort(s => String(s.name), "asc").array()) {
    const name = String(s.name || s.file.name);
    const disp = (String(s.emoji || "") + " " + name).trim();
    const cTasks = TASKS.where(t => t.subject === name && t.status !== "готово").array();
    const cCons  = CONS.where(c => c.subject === name).array();
    const el = grid.createEl("div");
    el.style.cssText = "background:" + C.card + ";border:1px solid " + C.border + ";border-radius:10px;padding:12px;cursor:pointer;transition:border-color .15s;";
    el.addEventListener("mouseenter", () => el.style.borderColor = C.accent);
    el.addEventListener("mouseleave", () => el.style.borderColor = C.border);
    el.addEventListener("click", () => app.workspace.openLinkText(s.file.path, "", false));
    el.createEl("div", { text: disp }).style.cssText = "font-size:14px;font-weight:700;color:" + C.text + ";margin-bottom:4px;";
    if (s.teacher) el.createEl("div", { text: s.teacher }).style.cssText = "font-size:11px;color:" + C.muted + ";margin-bottom:8px;";
    el.createEl("div", { text: "📓 " + cCons.length + " конспектов · 🎯 " + cTasks.length + " ДЗ" }).style.cssText = "font-size:11px;color:" + (cTasks.length ? C.accent : C.muted) + ";";
}

div("font-size:11px;font-weight:600;color:" + C.muted + ";text-transform:uppercase;letter-spacing:0.08em;margin:4px 0 8px;").textContent = "Дедлайны";
const dCard = div("background:" + C.card + ";border:1px solid " + C.border + ";border-radius:10px;padding:10px 14px;margin-bottom:14px;");
const sorted = activeTasks.sort((a, b) => { const x = toDate(a.due), y = toDate(b.due); if (!x) return 1; if (!y) return -1; return x < y ? -1 : (x > y ? 1 : 0); }).slice(0, 12);
if (sorted.length === 0) {
    dCard.createEl("div", { text: "Активных заданий нет — красота." }).style.cssText = "color:" + C.muted + ";font-size:13px;padding:4px 0;";
} else {
    for (const t of sorted) {
        const row = dCard.createEl("div");
        row.style.cssText = "display:flex;align-items:center;gap:8px;padding:5px 0;border-bottom:1px solid " + C.border + ";";
        const d = toDate(t.due);
        const late = d && d < TODAY;
        row.createEl("span", { text: d ? d.toFormat("dd.MM") : "—" }).style.cssText = "font-size:12px;color:" + (late ? C.accentLight : C.muted) + ";font-weight:" + (late ? 700 : 400) + ";flex-shrink:0;width:38px;";
        const a = row.createEl("a", { text: t.task });
        a.style.cssText = "color:" + C.text + ";text-decoration:none;font-size:13px;flex:1;";
        a.addEventListener("click", (e) => { e.preventDefault(); app.workspace.openLinkText(t.file.path, "", false); });
        if (t.subject) row.appendChild(badge((EMO_OF[t.subject] || "") + t.subject, "#3a3a3a"));
    }
}

div("font-size:11px;font-weight:600;color:" + C.muted + ";text-transform:uppercase;letter-spacing:0.08em;margin:4px 0 8px;").textContent = "Последние конспекты";
const cCard = div("background:" + C.card + ";border:1px solid " + C.border + ";border-radius:10px;padding:10px 14px;");
const recent = CONS.sort(c => toDate(c.date), "desc").slice(0, 6).array();
if (recent.length === 0) {
    cCard.createEl("div", { text: "Конспектов пока нет. Открой карточку предмета и нажми «＋ Конспект»." }).style.cssText = "color:" + C.muted + ";font-size:13px;padding:4px 0;";
} else {
    for (const c of recent) {
        const row = cCard.createEl("div");
        row.style.cssText = "display:flex;align-items:center;gap:8px;padding:5px 0;border-bottom:1px solid " + C.border + ";";
        row.createEl("span", { text: iso(c.date).slice(5) }).style.cssText = "font-size:11px;color:" + C.muted + ";flex-shrink:0;";
        const a = row.createEl("a", { text: (c.subject ? c.subject + " — " : "") + (c.тема || c.file.name) });
        a.style.cssText = "color:" + C.text + ";text-decoration:none;font-size:13px;flex:1;";
        a.addEventListener("click", (e) => { e.preventDefault(); app.workspace.openLinkText(c.file.path, "", false); });
    }
}
