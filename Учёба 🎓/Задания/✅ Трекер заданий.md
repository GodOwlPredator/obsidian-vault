# 🎯 ДЗ и дедлайны

```dataviewjs
const C = {
    accent: "#8f0000", accentLight: "#b30000", card: "#1e1e1e", border: "#2e2e2e",
    text: "#e0e0e0", muted: "#777", green: "#2d7a2d", yellow: "#6a6000", empty: "#2a2a2a",
};
const TODAY = dv.date("today");
const SUBJ  = dv.pages().where(s => s.type === "предмет" && s.name && !s.file.path.includes("/Templates/")).sort(s => String(s.name), "asc").array();
const EMO_OF = {};
for (const s of SUBJ) EMO_OF[String(s.name || s.file.name)] = String(s.emoji || "");

function toDate(d) {
    if (!d) return null;
    if (typeof d === "object" && typeof d.toFormat === "function") return d;
    try { return dv.date(String(d)); } catch (e) { return null; }
}
function iso(d) { const x = toDate(d); return x ? x.toFormat("yyyy-MM-dd") : String(d || ""); }
function fmt(d) { const x = toDate(d); return x ? x.toFormat("dd.MM.yyyy") : String(d || ""); }
function safe(s) { return String(s || "").replace(/[\\/:*?"<>|]/g, "-").trim().slice(0, 40); }

async function setValue(path, key, val) {
    const tf = app.vault.getAbstractFileByPath(path);
    if (!tf) return;
    let content = await app.vault.read(tf);
    const re = new RegExp("^(" + key + ":\\s*).*$", "m");
    content = re.test(content)
        ? content.replace(re, "$1" + val)
        : content.replace(/^---\n/, "---\n" + key + ": " + val + "\n");
    await app.vault.modify(tf, content);
}

function uniquePath(folder, base) {
    let p = folder + base + ".md", i = 2;
    while (app.vault.getAbstractFileByPath(p)) p = folder + base + " (" + i + ").md", i++;
    return p;
}

async function createTask(subject, task, dueStr) {
    const t = app.vault.getAbstractFileByPath("Templates/📝 Шаблон задания.md");
    if (!t) { new Notice("Нет шаблона"); return; }
    let s = await app.vault.read(t);
    s = s.split("@subject@").join(subject || "?")
         .split("@task@").join(task.trim() || "Новое задание")
         .split("TPL_DATE_ISO").join(dueStr || iso(TODAY))
         .split("TPL_DATE").join(fmt(toDate(dueStr) || TODAY));
    const path = uniquePath("Учёба 🎓/Задания/", (dueStr || iso(TODAY)) + " " + safe(subject) + " — " + safe(task));
    await app.vault.create(path, s);
    new Notice("Задание создано");
}

const root = dv.el("div", "");
root.style.cssText = "font-family:sans-serif;color:" + C.text;

const all = dv.pages().where(t => t.type === "задание" && t.task && !t.file.path.includes("/Templates/")).array();
const active = all.filter(t => t.status !== "готово");
const overdue = active.filter(t => { const d = toDate(t.due); return d && d < TODAY; });
const done = all.filter(t => t.status === "готово");

const stats = root.createEl("div");
stats.style.cssText = "display:flex;gap:12px;flex-wrap:wrap;margin-bottom:14px;";
function chip(v, l, color) {
    const c = stats.createEl("div");
    c.style.cssText = "background:" + C.card + ";border:1px solid " + C.border + ";border-radius:8px;padding:8px 14px;text-align:center;";
    c.createEl("div", { text: String(v) }).style.cssText = "font-size:20px;font-weight:700;color:" + (color || C.accent) + ";line-height:1.2;";
    c.createEl("div", { text: l }).style.cssText = "font-size:10px;color:" + C.muted + ";text-transform:uppercase;letter-spacing:0.05em;";
}
chip(active.length, "в работе", C.yellow);
chip(overdue.length, "просрочено", C.accentLight);
chip(done.length, "выполнено", C.green);

const form = root.createEl("div");
form.style.cssText = "display:flex;gap:8px;flex-wrap:wrap;align-items:center;background:" + C.card + ";border:1px solid " + C.border + ";border-radius:10px;padding:10px 14px;margin-bottom:14px;";
const sel = form.createEl("select");
sel.style.cssText = "background:" + C.empty + ";color:" + C.text + ";border:1px solid " + C.border + ";border-radius:6px;padding:6px;font-size:13px;";
const opt0 = sel.createEl("option", { text: "Предмет…" }); opt0.value = "";
for (const s of SUBJ) {
    const nm = String(s.name || s.file.name);
    const o = sel.createEl("option", { text: (String(s.emoji || "") + " " + nm).trim() });
    o.value = nm;
}
const inp = form.createEl("input", { type: "text", placeholder: "Что сделать?" });
inp.style.cssText = "flex:1;min-width:180px;background:" + C.empty + ";color:" + C.text + ";border:1px solid " + C.border + ";border-radius:6px;padding:6px;font-size:13px;";
inp.addEventListener("keydown", (e) => { if (e.key === "Enter") add(); });
const dueInp = form.createEl("input", { type: "date" });
dueInp.style.cssText = "background:" + C.empty + ";color:" + C.text + ";border:1px solid " + C.border + ";border-radius:6px;padding:6px;font-size:13px;";
dueInp.value = iso(TODAY);
const addBtn = form.createEl("button", { text: "＋ Добавить" });
addBtn.style.cssText = "padding:6px 14px;border-radius:6px;border:1px solid " + C.accent + ";background:" + C.accent + ";color:#fff;font-size:13px;cursor:pointer;";
async function add() {
    if (!sel.value || !inp.value) { new Notice("Выбери предмет и напиши задачу"); return; }
    await createTask(sel.value, inp.value, dueInp.value || iso(TODAY));
    inp.value = "";
}

function section(title) {
    const h = root.createEl("div");
    h.style.cssText = "font-size:11px;font-weight:600;color:" + C.muted + ";text-transform:uppercase;letter-spacing:0.08em;margin:12px 0 8px;";
    h.textContent = title;
    const box = root.createEl("div");
    box.style.cssText = "background:" + C.card + ";border:1px solid " + C.border + ";border-radius:10px;padding:8px 14px;";
    return box;
}
function row(box, t, late) {
    const r = box.createEl("div");
    r.style.cssText = "display:flex;align-items:center;gap:8px;padding:5px 0;border-bottom:1px solid " + C.border + ";opacity:" + (t.status === "готово" ? ".5" : "1") + ";";
    const cb = r.createEl("button", { text: t.status === "готово" ? "✓" : "○" });
    cb.style.cssText = "width:18px;height:18px;border-radius:4px;border:1px solid " + (t.status === "готово" ? C.green : C.border) + ";background:" + (t.status === "готово" ? C.green : "transparent") + ";color:#fff;font-size:11px;cursor:pointer;flex-shrink:0;";
    cb.addEventListener("click", () => setValue(t.file.path, "status", t.status === "готово" ? "в работе" : "готово"));
    const a = r.createEl("a", { text: t.task });
    a.style.cssText = "color:" + (t.status === "готово" ? C.muted : C.text) + ";text-decoration:none;font-size:13px;flex:1;";
    a.addEventListener("click", (e) => { e.preventDefault(); app.workspace.openLinkText(t.file.path, "", false); });
    if (t.subject) r.createEl("span", { text: (EMO_OF[t.subject] || "") + t.subject }).style.cssText = "font-size:10px;color:" + C.muted + ";";
    const d = toDate(t.due);
    if (d) r.createEl("span", { text: d.toFormat("dd.MM") }).style.cssText = "font-size:11px;color:" + (late ? C.accentLight : C.muted) + ";font-weight:" + (late ? 700 : 400) + ";";
}

const byDue = (a, b) => { const x = toDate(a.due), y = toDate(b.due); if (!x) return 1; if (!y) return -1; return x < y ? -1 : (x > y ? 1 : 0); };

const lateBox = section("Просрочено");
const lateList = active.filter(t => { const d = toDate(t.due); return d && d < TODAY; }).sort(byDue);
if (lateList.length === 0) lateBox.createEl("div", { text: "Нет — всё в срок." }).style.cssText = "color:" + C.muted + ";font-size:13px;padding:4px 0;";
else for (const t of lateList) row(lateBox, t, true);

const soonBox = section("Ближайшие дедлайны (7 дней)");
const soonList = active.filter(t => { const d = toDate(t.due); return d && d >= TODAY && d <= TODAY.plus({ days: 7 }); }).sort(byDue);
if (soonList.length === 0) soonBox.createEl("div", { text: "Ничего на неделе." }).style.cssText = "color:" + C.muted + ";font-size:13px;padding:4px 0;";
else for (const t of soonList) row(soonBox, t, false);

const restBox = section("Остальные");
const restList = active.filter(t => { const d = toDate(t.due); return !d || d > TODAY.plus({ days: 7 }); }).sort(byDue);
if (restList.length === 0) restBox.createEl("div", { text: "Пусто." }).style.cssText = "color:" + C.muted + ";font-size:13px;padding:4px 0;";
else for (const t of restList) row(restBox, t, false);

const doneBox = section("Выполненные");
if (done.length === 0) doneBox.createEl("div", { text: "Пока ничего не сдано." }).style.cssText = "color:" + C.muted + ";font-size:13px;padding:4px 0;";
else for (const t of done.slice(0, 10)) row(doneBox, t, false);
```
