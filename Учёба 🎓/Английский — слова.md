


# Английский — слова

**5 слов в день** из того, что и так смотришь — видео или статьи по электронике, 3D, sim racing. Слово вписываешь руками, перевод подставляет кнопка «Перевести пустые» (можно и самому).

Тренировка спрашивает в обе стороны: показывает английское слово — вводишь русский перевод, и наоборот. Режим переключается в карточке («Микс» по умолчанию), опечатка в одну букву прощается.

Заполняешь только первые три колонки. «Добавлено», «Уровень» и «Следующий повтор» — служебные, их считает тренировка.

  function parseWords(text) {

    if (a === b) return 0;
    if (!a.length || !b.length) return Math.max(a.length, b.length);
    let prev = Array.from({ length: b.length + 1 }, (_, i) => i);
    for (let i = 1; i <= a.length; i++) {

      for (let j = 1; j <= b.length; j++) {
        cur[j] = Math.min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1));

    btnTranslate.textContent = n ? `Перевести пустые (${n})` : "Переводы заполнены";
    btnTranslate.style.opacity = n ? "1" : "0.55";
    btnTranslate.style.cursor = n ? "pointer" : "default";
  }

  btnTranslate.onclick = async () => {
    if (btnTranslate.disabled) return;
    btnTranslate.disabled = true;
    btnTranslate.textContent = "Перевожу…";
    const n = await translateEmpty();
    await refresh();
    btnTranslate.textContent = n ? `Готово: ${n}` : "Не вышло — проверь сеть";
    setTimeout(refresh, 1800);
  };

  setButton(initialWords);
  setTranslateButton(initialWords);
  boot(8);
})();
