// ─── СЕТКА КНИГ ───────────────────────────────────────────────────────────────
const gridEl = root.createEl("div", { cls: "bl-grid" });

function getSortedFiltered() {
    let list = allBooks.array();
    if (activeStatus !== "все") list = list.filter(b => b.status === activeStatus);
    if (activeGenre  !== "все") list = list.filter(b => String(b.genre) === activeGenre);

    list.sort((a, b) => {
        if (activeSort === "rating") return (Number(b.rating) || 0) - (Number(a.rating) || 0);
        if (activeSort === "title")  return String(a.title || a.file.name).localeCompare(String(b.title || b.file.name));
        if (activeSort === "pages")  return (Number(b.pages) || 0) - (Number(a.pages) || 0);
        // date
        const da = toDate(a.date_finished) || toDate(a.date_started) || a.file.mtime;
        const db = toDate(b.date_finished) || toDate(b.date_started) || b.file.mtime;
        if (!da && !db) return 0;
        if (!da) return 1;
        if (!db) return -1;
        return db - da;
    });
    return list;
}

function renderGrid() {
    gridEl.innerHTML = "";
    const list = getSortedFiltered();

    if (list.length === 0) {
        gridEl.createEl("div", { cls: "bl-empty", text: "В этой категории пусто." });
        return;
    }

    for (const b of list) {
        const card = gridEl.createEl("div", { cls: "bl-card" });
        card.addEventListener("click", () => app.workspace.openLinkText(b.file.path, "", false));

        // Обложка
        const coverVal = b.cover || "";
        if (coverVal) {
            const cp = String(coverVal).replace(/^\[\[|\]\]$/g, "");
            const tf = app.vault.getAbstractFileByPath(cp)
                || app.metadataCache.getFirstLinkpathDest(cp, b.file.path);
            if (tf) {
                const img = card.createEl("img", { cls: "bl-cover" });
                img.src = app.vault.getResourcePath(tf);
                img.alt = b.title || b.file.name;
            } else {
                card.createEl("div", { cls: "bl-cover-ph", text: "📖" });
            }
        } else {
            card.createEl("div", { cls: "bl-cover-ph", text: "📖" });
        }

        const info = card.createEl("div", { cls: "bl-info" });
        info.createEl("div", { text: b.title || b.file.name, cls: "bl-title" });
        info.createEl("div", { text: b.author || "—", cls: "bl-author" });

        const bottom = info.createEl("div", {});
        bottom.style.cssText = "display: flex; justify-content: space-between; align-items: center;";
        if (b.status) {
            const badge = bottom.createEl("span", { text: STATUS_LABELS[b.status] || b.status, cls: "bl-badge" });
            badge.style.background = STATUS_COLORS[b.status] || C.muted;
        }
        if (Number(b.rating) > 0) {
            bottom.createEl("span", { text: "★".repeat(b.rating), cls: "bl-stars" });
        }

        // Мини-прогресс (для книг в процессе чтения)
        if (b.status === "читаю" && Number(b.pages) > 0) {
            const pct = Math.min(100, Math.round((Number(b.pages_read)||0) / Number(b.pages) * 100));
            const pw = info.createEl("div", { cls: "bl-progress-mini" });
            pw.createEl("div", { cls: "bl-progress-mini-fill" }).style.width = pct + "%";
        }
    }
}
