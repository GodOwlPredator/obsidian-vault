# 📖 Граф Монте-Кристо

```dataviewjs
const C = {
    accent:      "#8f0000",
    accentLight: "#b30000",
    accentDim:   "#3d0000",
    accentHover: "#6a0000",
    empty:       "#2a2a2a",
    emptyBorder: "#444",
    card:        "#1e1e1e",
    border:      "#2e2e2e",
    text:        "#e0e0e0",
    muted:       "#777",
    green:       "#2d7a2d",
    yellow:      "#6a6000",
};

function toDate(d) {
    if (!d) return null;
    if (typeof d === "object" && typeof d.toFormat === "function") return d;
    try { return dv.date(String(d)); } catch(e) { return null; }
}

const file       = dv.current().file;
const p          = dv.current();
const rating     = Number(p.rating)     || 0;
const pagesTotal = Number(p.pages)      || 0;
const pagesRead  = Number(p.pages_read) || 0;

async function setValue(key, val) {
    const tf = app.vault.getAbstractFileByPath(file.path);
    const content = await app.vault.read(tf);
    const regex = new RegExp(`^(${key}:\\s*).*$`, "m");
    const updated = regex.test(content)
        ? content.replace(regex, `$1${val}`)
        : content.startsWith("---")
            ? content.replace(/^---\n/, `---\n${key}: ${val}\n`)
            : `---\n${key}: ${val}\n---\n` + content;
    await app.vault.modify(tf, updated);
}

const root = dv.el("div", "", {});
root.style.cssText = `font-family: sans-serif; color: ${C.text};`;

// ── Карточка: обложка + мета ───────────────────────────────────────────────────
const card = root.createEl("div", {});
card.style.cssText = `display: flex; gap: 20px; background: ${C.card}; border: 1px solid ${C.border}; border-radius: 12px; padding: 16px; margin-bottom: 12px; align-items: flex-start; flex-wrap: wrap;`;

// Обложка
const coverWrap = card.createEl("div", {});
coverWrap.style.cssText = "flex-shrink: 0;";
const cover = String(p.cover || "");
if (cover) {
    const coverPath = cover.replace(/^\[\[|\]\]$/g, "");
    const tf2 = app.vault.getAbstractFileByPath(coverPath)
        || app.metadataCache.getFirstLinkpathDest(coverPath, file.path);
    if (tf2) {
        const img = coverWrap.createEl("img", {});
        img.src = app.vault.getResourcePath(tf2);
        img.style.cssText = "width: 110px; height: 165px; object-fit: cover; border-radius: 8px; display: block;";
    } else {
        const ph = coverWrap.createEl("div", {});
        ph.style.cssText = `width: 110px; height: 165px; background: linear-gradient(135deg, ${C.accentDim}, #111); border-radius: 8px; display: flex; align-items: center; justify-content: center; font-size: 44px;`;
        ph.textContent = "📖";
    }
} else {
    const ph = coverWrap.createEl("div", {});
    ph.style.cssText = `width: 110px; height: 165px; background: linear-gradient(135deg, ${C.accentDim}, #111); border-radius: 8px; display: flex; align-items: center; justify-content: center; font-size: 44px;`;
    ph.textContent = "📖";
}

// Правая часть: мета
const meta = card.createEl("div", {});
meta.style.cssText = "flex: 1; min-width: 180px; display: flex; flex-direction: column; gap: 10px;";

// Заголовок + автор + теги
const topBlock = meta.createEl("div", {});
topBlock.createEl("h2", { text: p.title || file.name }).style.cssText = `margin: 0 0 3px; font-size: 18px; color: ${C.text};`;
topBlock.createEl("div", { text: p.author || "Неизвестный автор" }).style.cssText = `font-size: 13px; color: ${C.muted};`;

const tagsRow = meta.createEl("div", {});
tagsRow.style.cssText = "display: flex; gap: 6px; flex-wrap: wrap;";
if (p.genre) {
    tagsRow.createEl("span", { text: "🏷 " + p.genre }).style.cssText = `background: ${C.empty}; color: ${C.muted}; border-radius: 5px; padding: 3px 9px; font-size: 11px; border: 1px solid ${C.emptyBorder};`;
}
if (pagesTotal > 0) {
    tagsRow.createEl("span", { text: pagesTotal + " стр." }).style.cssText = `background: ${C.empty}; color: ${C.muted}; border-radius: 5px; padding: 3px 9px; font-size: 11px; border: 1px solid ${C.emptyBorder};`;
}

// Переключатель статуса
const statusRow = meta.createEl("div", {});
statusRow.style.cssText = "display: flex; gap: 5px; flex-wrap: wrap;";
const STATUS_COLORS = { "читаю": C.accent, "прочитано": C.green, "хочу": C.yellow, "брошено": "#555" };
const STATUS_LABELS = { "хочу": "📋 Хочу", "читаю": "📖 Читаю", "прочитано": "✓ Прочитано", "брошено": "✕ Брошено" };
for (const s of ["хочу", "читаю", "прочитано", "брошено"]) {
    const active = p.status === s;
    const btn = statusRow.createEl("button", { text: STATUS_LABELS[s] });
    btn.style.cssText = `padding: 4px 10px; border-radius: 5px; border: 1px solid ${active ? STATUS_COLORS[s] : C.emptyBorder}; background: ${active ? STATUS_COLORS[s] : C.empty}; color: ${active ? "#fff" : C.muted}; font-size: 11px; font-weight: ${active ? "700" : "400"}; cursor: pointer; transition: background-color .15s, border-color .15s, color .15s;`;
    btn.addEventListener("mouseenter", () => { if (!active) { btn.style.borderColor = STATUS_COLORS[s]; btn.style.color = C.text; } });
    btn.addEventListener("mouseleave", () => { if (!active) { btn.style.borderColor = C.emptyBorder; btn.style.color = C.muted; } });
    btn.addEventListener("click", async () => { await setValue("status", s); });
}

// Звёздный рейтинг
const ratingRow = meta.createEl("div", {});
ratingRow.style.cssText = "display: flex; align-items: center; gap: 4px;";
ratingRow.createEl("span", { text: "Оценка:" }).style.cssText = `font-size: 12px; color: ${C.muted}; margin-right: 4px;`;
for (let i = 1; i <= 5; i++) {
    const star = ratingRow.createEl("span", { text: i <= rating ? "★" : "☆" });
    star.style.cssText = `font-size: 22px; cursor: pointer; color: ${i <= rating ? C.accent : C.muted}; transition: color 0.1s, transform 0.1s;`;
    star.addEventListener("mouseenter", () => { star.style.transform = "scale(1.2)"; star.style.color = C.accent; });
    star.addEventListener("mouseleave", () => { star.style.transform = "scale(1)"; star.style.color = i <= rating ? C.accent : C.muted; });
    star.addEventListener("click", async () => { await setValue("rating", i === rating ? 0 : i); });
}

// Ссылка на библиотеку
const libLink = meta.createEl("a", { text: "← Вся библиотека" });
libLink.style.cssText = `font-size: 11px; color: ${C.accent}; text-decoration: none; padding: 2px 8px; border: 1px solid ${C.accent}; border-radius: 4px; align-self: flex-start;`;
libLink.addEventListener("click", (e) => {
    e.preventDefault();
    const lib = dv.pages().where(f => f.cssclasses && f.cssclasses.includes("book-library")).first();
    if (lib) app.workspace.openLinkText(lib.file.path, "", false);
});

// ── Прогресс чтения ───────────────────────────────────────────────────────────
if (pagesTotal > 0) {
    const pct = Math.min(100, Math.round(pagesRead / pagesTotal * 100));
    const progCard = root.createEl("div", {});
    progCard.style.cssText = `background: ${C.card}; border: 1px solid ${C.border}; border-radius: 10px; padding: 14px 18px; margin-bottom: 12px;`;

    // Заголовок прогресс-карточки
    const ph = progCard.createEl("div", {});
    ph.style.cssText = "display: flex; justify-content: space-between; align-items: center; margin-bottom: 10px;";
    ph.createEl("span", { text: "Прогресс чтения" }).style.cssText = `font-size: 11px; font-weight: 600; color: ${C.muted}; text-transform: uppercase; letter-spacing: 0.07em;`;
    ph.createEl("span", { text: `${pagesRead} / ${pagesTotal} стр. · ${pct}%` }).style.cssText = `font-size: 13px; font-weight: 700; color: ${pct >= 100 ? C.green : C.accent};`;

    // Прогресс-бар
    const barWrap = progCard.createEl("div", {});
    barWrap.style.cssText = `height: 10px; background: ${C.empty}; border-radius: 5px; overflow: hidden; margin-bottom: 12px;`;
    barWrap.createEl("div", {}).style.cssText = `height: 100%; width: ${pct}%; background: ${pct >= 100 ? C.green : C.accent}; border-radius: 5px; transition: width 0.3s;`;

    // Кнопки изменения страниц
    const ctrlRow = progCard.createEl("div", {});
    ctrlRow.style.cssText = "display: flex; gap: 6px; align-items: center; flex-wrap: wrap; margin-bottom: 10px;";
    ctrlRow.createEl("span", { text: "Прочитал:" }).style.cssText = `font-size: 11px; color: ${C.muted}; margin-right: 2px;`;

    for (const delta of [-50, -10, +10, +50]) {
        const btn = ctrlRow.createEl("button", { text: (delta > 0 ? "+" : "") + delta });
        btn.style.cssText = `padding: 4px 10px; border-radius: 5px; border: 1px solid ${C.emptyBorder}; background: ${C.empty}; color: ${delta > 0 ? C.accent : C.muted}; font-size: 12px; font-weight: 600; cursor: pointer; transition: background-color .1s, border-color .1s, color .1s;`;
        btn.addEventListener("mouseenter", () => { btn.style.borderColor = C.accent; btn.style.color = C.accent; });
        btn.addEventListener("mouseleave", () => { btn.style.borderColor = C.emptyBorder; btn.style.color = delta > 0 ? C.accent : C.muted; });
        btn.addEventListener("click", async () => {
            await setValue("pages_read", Math.max(0, Math.min(pagesTotal, pagesRead + delta)));
        });
    }

    if (pct < 100) {
        const doneBtn = ctrlRow.createEl("button", { text: "✓ Дочитал!" });
        doneBtn.style.cssText = `padding: 4px 12px; border-radius: 5px; border: 1px solid ${C.green}; background: ${C.green}; color: #fff; font-size: 12px; font-weight: 700; cursor: pointer; margin-left: 4px;`;
        doneBtn.addEventListener("click", async () => {
            await setValue("pages_read", pagesTotal);
            await setValue("status", "прочитано");
            if (!p.date_finished) await setValue("date_finished", dv.date("today").toFormat("yyyy-MM-dd"));
        });
    }

    // Статистика скорости чтения
    const dateStart = toDate(p.date_started);
    const dateEnd   = toDate(p.date_finished);

    if (dateStart && pagesRead > 0 && p.status === "читаю") {
        const today = dv.date("today");
        const days  = Math.max(1, Math.round(today.diff(dateStart, "days").days));
        const speed = (pagesRead / days).toFixed(1);
        const remaining = pagesTotal - pagesRead;
        const etaDays = speed > 0 ? Math.ceil(remaining / parseFloat(speed)) : null;

        const speedRow = progCard.createEl("div", {});
        speedRow.style.cssText = `display: flex; gap: 16px; flex-wrap: wrap; font-size: 12px; color: ${C.muted}; padding: 8px 0; border-top: 1px solid ${C.border}; border-bottom: 1px solid ${C.border}; margin-bottom: 8px;`;
        speedRow.createEl("span", { text: `📅 Читаю ${days} дн.` });
        speedRow.createEl("span", { text: `⚡ ${speed} стр./день` });
        if (etaDays !== null && etaDays > 0) speedRow.createEl("span", { text: `🏁 Осталось ~${etaDays} дн.` });
    }

    // Даты
    if (dateStart || dateEnd) {
        const datesRow = progCard.createEl("div", {});
        datesRow.style.cssText = `display: flex; gap: 20px; flex-wrap: wrap; font-size: 12px; color: ${C.muted};`;
        if (dateStart) datesRow.createEl("span", { text: "🗓 Начал: " + dateStart.toFormat("dd.MM.yyyy") });
        if (dateEnd)   datesRow.createEl("span", { text: "✅ Закончил: " + dateEnd.toFormat("dd.MM.yyyy") });
        if (dateStart && dateEnd) {
            const totalDays = Math.round(dateEnd.diff(dateStart, "days").days);
            datesRow.createEl("span", { text: `⏱ За ${totalDays} дней` }).style.color = C.accent;
        }
    }
}
```
