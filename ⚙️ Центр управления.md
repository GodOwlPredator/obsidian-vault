  const refreshPrices = async () => {
    try {
      const f = app.vault.getAbstractFileByPath(".dns-prices-status");
      if (!f) { statusEl.textContent = "цены: нет данных"; return; }
      const j = JSON.parse(await app.vault.read(f));
      const ts = j.ts ? new Date(j.ts) : null;
      let rel = "";
      if (ts) { const m = Math.floor((Date.now()-ts)/60000); rel = m < 1 ? "только что" : m < 60 ? m + " мин назад" : Math.floor(m/60) + " ч назад"; }
      statusEl.textContent = "цены: " + rel + " · " + (j.got||0) + "/" + (j.total||0) + (j.changed ? " (+" + j.changed + ")" : "");
    } catch(e) { statusEl.textContent = "цены: нет данных"; }
  };
  refreshPrices();
  setInterval(refreshPrices, 30000);

  c.addEventListener("click", async () => {
    try {
      const flag = app.vault.getAbstractFileByPath(".dns-prices-request");
      if (!flag) await app.vault.create(".dns-prices-request", String(Date.now()));
      statusEl.textContent = "запускаю…";
      new Notice("🔄 Обновляю цены с DNS (займёт ~1–2 мин)");
    } catch (e) { new Notice("Не удалось запустить: " + e); }
  });
}

// автозапуск рекордера: после старта Obsidian открыть его в правой панели,
// если он ещё не открыт (разового открытия хватает — панель сохраняется)
(function ensureRecorder() {
  if (window.__cc_autoStart) return;
  window.__cc_autoStart = true;
  const open = async () => {
    try {
      const file = app.vault.getAbstractFileByPath("Голосовой рекодер 🎙️/Голосовой рекодер 🎙️.md");
      if (!file) return;
      const already = app.workspace.getLeavesOfType("markdown").some(l => {
        const f = l.view && l.view.file;
        return f && f.path === file.path;
      });
      if (already) return;
      const leaf = app.workspace.getRightLeaf(false);
      if (leaf) await leaf.openFile(file);
    } catch (e) { /* тихо */ }
  };
  if (app.workspace.layoutReady) open();
  else app.workspace.onLayoutReady(open);
})();
