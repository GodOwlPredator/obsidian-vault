```dataviewjs
// ── Кнопка добавления новой даты ─────────────────────────────────
const btn = dv.el("button", "＋ Добавить запись за сегодня");
btn.style.cssText = [
  "cursor:pointer",
  "background:#1e1e1e",
  "border:1px solid #2e2e2e",
  "border-radius:8px",
  "color:#555",
  "font-size:12px",
  "padding:5px 14px",
  "margin-bottom:4px",
  "transition:border-color 0.15s,color 0.15s",
  "outline:none",
].join(";");
btn.addEventListener("mouseenter", () => {
  btn.style.borderColor = "#8f0000";
  btn.style.color = "#e0e0e0";
});
btn.addEventListener("mouseleave", () => {
  btn.style.borderColor = "#2e2e2e";
  btn.style.color = "#555";
});

btn.addEventListener("click", async () => {
  const today = moment().format("DD.MM.YYYY");
  // Не пишем сегодняшнюю дату в чужой месячный файл
  const RU_M = {"январь":"01","февраль":"02","март":"03","апрель":"04","май":"05","июнь":"06",
                "июль":"07","август":"08","сентябрь":"09","октябрь":"10","ноябрь":"11","декабрь":"12"};
  const fm = dv.current().file;
  const fMonth = RU_M[fm.name.toLowerCase()];
  const fYear = (fm.path.match(/\/(\d{4})\//) || [])[1];
  if (fMonth && `${fMonth}.${fYear}` !== moment().format("MM.YYYY")) {
    new Notice(`📅 Сегодня ${today}, а этот файл — за ${fm.name} ${fYear}. Открой нужный месяц.`);
    return;
  }
  const entry = `\n<font color="#8f0000">${today}</font>\n***\n\n- `;

  const file = app.vault.getAbstractFileByPath(dv.current().file.path);
  const content = await app.vault.read(file);

  if (content.includes(`>${today}</font>`)) {
    new Notice(`📅 ${today} уже есть в заметке`);
    return;
  }

  await app.vault.modify(file, content.trimEnd() + entry);
  new Notice(`✅ Добавлена запись за ${today}`);
});
```
