# 🔧 Название инструмента

```dataviewjs
const C = { accent:"#8f0000",accentLight:"#b30000",accentDim:"#3d0000",card:"#1e1e1e",border:"#2e2e2e",text:"#e0e0e0",muted:"#777",empty:"#2a2a2a",emptyBorder:"#444",green:"#2d7a2d",yellow:"#6a6000" };

function toArr(v) {
    if (v == null) return [];
    if (Array.isArray(v)) return v.map(x => String(x));
    if (typeof v === "object" && typeof v.array === "function") return v.array().map(x => String(x));
    return [String(v)];
}

const p = dv.current(), file = dv.current().file;
const rating = Number(p.оценка) || 0;

async function setValue(key, val) {
    const tf = app.vault.getAbstractFileByPath(file.path);
    const content = await app.vault.read(tf);
    const regex = new RegExp(`^(${key}:\\s*).*$`, "m");
    const updated = regex.test(content)
        ? content.replace(regex, `$1${val}`)
        : content.startsWith("---")
            ? content.replace(/^---\n/, `---\n${key}: ${val}\n`)
            : `---\n${key}: ${val}\n---\n` + content;
    await app.vault.modify(tf, updated);
}

const root = dv.el("div", "", {});
root.style.cssText = `font-family:sans-serif;color:${C.text};`;
const card = root.createEl("div", {});
card.style.cssText = `background:${C.card};border:1px solid ${C.border};border-left:4px solid ${C.green};border-radius:10px;padding:16px 20px;margin-bottom:12px;`;

card.createEl("h2", { text: "🔧 " + (p.title || file.name) }).style.cssText = `margin:0 0 10px;font-size:18px;color:${C.text};`;

// Тип
if (p.тип) {
    card.createEl("span", { text: String(p.тип) }).style.cssText = `background:${C.empty};color:${C.muted};border-radius:5px;padding:3px 9px;font-size:11px;border:1px solid ${C.emptyBorder};display:inline-block;margin-bottom:10px;`;
}

// Рейтинг звёздами (зелёный цвет для инструментов)
const rr = card.createEl("div", {});
rr.style.cssText = "display:flex;align-items:center;gap:4px;margin-bottom:10px;";
rr.createEl("span", { text: "Оценка:" }).style.cssText = `font-size:12px;color:${C.muted};margin-right:4px;`;
for (let i = 1; i <= 5; i++) {
    const star = rr.createEl("span", { text: i <= rating ? "★" : "☆" });
    star.style.cssText = `font-size:20px;cursor:pointer;color:${i <= rating ? C.green : C.muted};transition:color 0.1s,transform 0.1s;`;
    star.addEventListener("mouseenter", () => { star.style.transform = "scale(1.2)"; star.style.color = C.green; });
    star.addEventListener("mouseleave", () => { star.style.transform = "scale(1)"; star.style.color = i <= rating ? C.green : C.muted; });
    star.addEventListener("click", async () => { await setValue("оценка", i === rating ? 0 : i); });
}

// Описание
if (p.описание) {
    const d = card.createEl("div", { text: String(p.описание) });
    d.style.cssText = `font-size:13px;color:${C.text};line-height:1.5;padding-top:10px;border-top:1px solid ${C.border};margin-bottom:10px;`;
}

// Ссылка
const linkStr = String(p.ссылка || "");
if (linkStr.startsWith("http")) {
    const a = card.createEl("a", { text: "🔗 Открыть сайт" });
    a.href = linkStr;
    a.target = "_blank";
    a.style.cssText = `display:inline-block;margin-bottom:10px;font-size:11px;color:${C.green};text-decoration:none;padding:2px 8px;border:1px solid ${C.green};border-radius:4px;`;
}

// Теги
const tags = toArr(p.теги).filter(Boolean);
if (tags.length > 0) {
    const tw = card.createEl("div", {});
    tw.style.cssText = "display:flex;gap:5px;flex-wrap:wrap;margin-bottom:8px;";
    for (const t of tags) {
        tw.createEl("span", { text: t }).style.cssText = `background:${C.empty};color:${C.muted};border-radius:4px;padding:2px 8px;font-size:11px;border:1px solid ${C.emptyBorder};`;
    }
}

// Назад
const back = card.createEl("a", { text: "← 3D Вики" });
back.style.cssText = `display:inline-block;font-size:11px;color:${C.accent};text-decoration:none;padding:2px 8px;border:1px solid ${C.accent};border-radius:4px;`;
back.addEventListener("click", (e) => {
    e.preventDefault();
    const home = dv.pages().where(f => f.cssclasses && toArr(f.cssclasses).includes("wiki-3d-home")).first();
    if (home) app.workspace.openLinkText(home.file.path, "", false);
});
```
