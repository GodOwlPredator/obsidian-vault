# 💡 Человек под дождём

```dataviewjs
const C = { accent:"#8f0000",accentLight:"#b30000",accentDim:"#3d0000",card:"#1e1e1e",border:"#2e2e2e",text:"#e0e0e0",muted:"#777",empty:"#2a2a2a",emptyBorder:"#444",green:"#2d7a2d",yellow:"#6a6000" };

// Безопасное преобразование в JS-массив строк
function toArr(v) {
    if (v == null) return [];
    if (Array.isArray(v)) return v.map(x => String(x));
    if (typeof v === "object" && typeof v.array === "function") return v.array().map(x => String(x));
    return [String(v)];
}

const p = dv.current(), file = dv.current().file;

async function setValue(key, val) {
    const tf = app.vault.getAbstractFileByPath(file.path);
    const content = await app.vault.read(tf);
    const regex = new RegExp(`^(${key}:\\s*).*$`, "m");
    const updated = regex.test(content)
        ? content.replace(regex, `$1${val}`)
        : content.startsWith("---")
            ? content.replace(/^---\n/, `---\n${key}: ${val}\n`)
            : `---\n${key}: ${val}\n---\n` + content;
    await app.vault.modify(tf, updated);
}

const root = dv.el("div", "", {});
root.style.cssText = `font-family:sans-serif;color:${C.text};`;
const card = root.createEl("div", {});
card.style.cssText = `background:${C.card};border:1px solid ${C.border};border-left:4px solid ${C.accent};border-radius:10px;padding:16px 20px;margin-bottom:12px;`;

// Заголовок
card.createEl("h2", { text: "💡 " + (p.title || file.name) }).style.cssText = `margin:0 0 12px;font-size:18px;color:${C.text};`;

// Статус
const STATUSES = ["новая", "в работе", "реализована", "отложена"];
const STATUS_C  = { "новая": C.accent, "в работе": "#4a90d9", "реализована": C.green, "отложена": C.muted };
const statusRow = card.createEl("div", {});
statusRow.style.cssText = "display:flex;gap:5px;flex-wrap:wrap;margin-bottom:8px;align-items:center;";
statusRow.createEl("span", { text: "Статус:" }).style.cssText = `font-size:11px;color:${C.muted};margin-right:2px;`;
for (const s of STATUSES) {
    const active = String(p.статус) === s;
    const btn = statusRow.createEl("button", { text: s });
    btn.style.cssText = `padding:3px 10px;border-radius:5px;border:1px solid ${active ? STATUS_C[s] : C.emptyBorder};background:${active ? STATUS_C[s] : C.empty};color:${active ? "#fff" : C.muted};font-size:11px;font-weight:${active ? "700" : "400"};cursor:pointer;transition: background-color .15s, border-color .15s, color .15s;`;
    btn.addEventListener("mouseenter", () => { if (!active) { btn.style.borderColor = STATUS_C[s]; btn.style.color = C.text; } });
    btn.addEventListener("mouseleave", () => { if (!active) { btn.style.borderColor = C.emptyBorder; btn.style.color = C.muted; } });
    btn.addEventListener("click", async () => { await setValue("статус", s); });
}

// Приоритет
const PRIORITIES = ["низкий", "средний", "высокий"];
const PR_C = { "низкий": C.muted, "средний": C.yellow, "высокий": C.accent };
const prRow = card.createEl("div", {});
prRow.style.cssText = "display:flex;gap:5px;flex-wrap:wrap;margin-bottom:10px;align-items:center;";
prRow.createEl("span", { text: "Приоритет:" }).style.cssText = `font-size:11px;color:${C.muted};margin-right:2px;`;
for (const pr of PRIORITIES) {
    const active = String(p.приоритет) === pr;
    const btn = prRow.createEl("button", { text: pr });
    btn.style.cssText = `padding:3px 10px;border-radius:5px;border:1px solid ${active ? PR_C[pr] : C.emptyBorder};background:${active ? PR_C[pr] : C.empty};color:${active ? "#fff" : C.muted};font-size:11px;font-weight:${active ? "700" : "400"};cursor:pointer;transition: background-color .15s, border-color .15s, color .15s;`;
    btn.addEventListener("mouseenter", () => { if (!active) { btn.style.borderColor = PR_C[pr]; btn.style.color = C.text; } });
    btn.addEventListener("mouseleave", () => { if (!active) { btn.style.borderColor = C.emptyBorder; btn.style.color = C.muted; } });
    btn.addEventListener("click", async () => { await setValue("приоритет", pr); });
}

// Описание
if (p.описание) {
    const d = card.createEl("div", { text: String(p.описание) });
    d.style.cssText = `font-size:13px;color:${C.text};line-height:1.5;padding-top:10px;border-top:1px solid ${C.border};margin-bottom:10px;`;
}

// Программы
const progs = toArr(p.программы).filter(Boolean);
if (progs.length > 0) {
    const pw = card.createEl("div", {});
    pw.style.cssText = "display:flex;gap:6px;flex-wrap:wrap;margin-bottom:8px;";
    pw.createEl("span", { text: "🔧" }).style.cssText = `font-size:12px;color:${C.muted};align-self:center;`;
    for (const prog of progs) {
        const t = pw.createEl("span", { text: prog });
        t.style.cssText = `background:${C.empty};color:${C.muted};border-radius:5px;padding:3px 9px;font-size:11px;border:1px solid ${C.emptyBorder};`;
    }
}

// Теги
const tags = toArr(p.теги).filter(Boolean);
if (tags.length > 0) {
    const tw = card.createEl("div", {});
    tw.style.cssText = "display:flex;gap:5px;flex-wrap:wrap;margin-bottom:8px;";
    for (const t of tags) {
        const tg = tw.createEl("span", { text: t });
        tg.style.cssText = `background:${C.empty};color:${C.muted};border-radius:4px;padding:2px 8px;font-size:11px;border:1px solid ${C.emptyBorder};`;
    }
}

// Назад
const back = card.createEl("a", { text: "← 3D Вики" });
back.style.cssText = `display:inline-block;font-size:11px;color:${C.accent};text-decoration:none;padding:2px 8px;border:1px solid ${C.accent};border-radius:4px;`;
back.addEventListener("click", (e) => {
    e.preventDefault();
    const home = dv.pages().where(f => f.cssclasses && toArr(f.cssclasses).includes("wiki-3d-home")).first();
    if (home) app.workspace.openLinkText(home.file.path, "", false);
});
```
