```dataviewjs
/* ═══ 3D Вики — главный хаб ═══ */
async function ensureD3Css() {
    if (document.getElementById("d3-css")) return;
    let loaded = "";
    try { loaded = getComputedStyle(document.documentElement).getPropertyValue("--d3-loaded").trim(); } catch (e) {}
    if (loaded === "1") return;
    let css = "";
    try { css = await app.vault.adapter.read(".obsidian/snippets/3d.css"); } catch (e) {}
    const s = document.createElement("style");
    s.id = "d3-css"; s.textContent = css; document.head.appendChild(s);
}
await ensureD3Css();

const BASE = dv.current().file.folder;

const KINDS = {
    "идея":       { icon: "💡", label: "Идея",       plural: "Идеи",       dir: "💡 Идеи",        tpl: "📝 Шаблон идеи.md" },
    "фишка":      { icon: "⚡", label: "Фишка",      plural: "Фишки",      dir: "Фишки",          tpl: "📝 Шаблон фишки.md" },
    "инструмент": { icon: "🔧", label: "Инструмент", plural: "Инструменты",dir: "🔧 Инструменты", tpl: "📝 Шаблон инструмента.md" },
    "референс":   { icon: "🖼", label: "Референс",   plural: "Референсы",  dir: "🎯 Референсы",   tpl: "📝 Шаблон референса.md" },
    "проект":     { icon: "📦", label: "Проект",     plural: "Проекты",    dir: "📦 Проекты",     tpl: "📝 Шаблон проекта.md" },
    "материал":   { icon: "🎨", label: "Материал",   plural: "Материалы",  dir: "🎨 Материалы",   tpl: "📝 Шаблон материала.md" },
    "ассет":      { icon: "🧰", label: "Ассет",      plural: "Ассеты",     dir: "🧰 Ассеты",      tpl: "📝 Шаблон ассета.md" },
    "обучение":   { icon: "🎓", label: "Обучение",   plural: "Обучение",   dir: "🎓 Обучение",    tpl: "📝 Шаблон обучения.md" },
    "челлендж":   { icon: "🏆", label: "Челлендж",   plural: "Челленджи",  dir: "🏆 Челленджи",   tpl: "📝 Шаблон челленджа.md" },
    "чеклист":    { icon: "✅", label: "Чеклист",    plural: "Чеклисты",   dir: "✅ Чеклисты",    tpl: "📝 Шаблон чеклиста.md" },
};
const ORDER = Object.keys(KINDS);
const COLORS = {
    "идея": "#a80000", "фишка": "#4a90d9", "инструмент": "#3f9d3f", "референс": "#c9a227",
    "проект": "#d13b3b", "материал": "#a855c9", "ассет": "#3f9d9d", "обучение": "#4a90d9",
    "челлендж": "#c9a227", "чеклист": "#3f9d3f",
};

function toArr(v) {
    if (v == null) return [];
    if (Array.isArray(v)) return v.map(x => String(x));
    if (typeof v === "object" && typeof v.array === "function") return v.array().map(x => String(x));
    return [String(v)];
}
function findPage(cls) {
    return dv.pages().where(x => x.cssclasses && toArr(x.cssclasses).includes(cls)).first();
}
function coverUrl(v, fromPath) {
    const cp = String(v || "").replace(/^!?\[\[|\]\]$/g, "").trim();
    if (!cp) return null;
    const f = app.vault.getAbstractFileByPath(cp) || app.metadataCache.getFirstLinkpathDest(cp, fromPath);
    return f ? app.vault.getResourcePath(f) : null;
}
function openNote(pathStr) { app.workspace.openLinkText(pathStr, "", false); }

/* ── Корень ─────────────────────────────────────────────────────── */
const root = dv.el("div", "", {});
root.className = "d3-root";

/* ── Панель добавления ──────────────────────────────────────────── */
const top = root.createEl("div", { cls: "d3-top" });
const addBtns = [];
for (const k of ORDER) {
    const K = KINDS[k];
    const b = top.createEl("button", { text: "＋ " + K.label, cls: "d3-btn d3-btn-accent" });
    b.title = "Создать: " + K.plural;
    b.addEventListener("click", () => wkForm(k));
    addBtns.push({ k, b });
}
top.createEl("div", { cls: "d3-top-sep" });
const navDefs = [
    ["🖼 Галерея работ", "gallery-home"],
    ["🧭 Пайплайн", "pipeline-3d"],
];
for (const [txt, cls] of navDefs) {
    const b = top.createEl("button", { text: txt, cls: "d3-btn" });
    b.addEventListener("click", () => {
        const pg = findPage(cls);
        if (pg) openNote(pg.file.path);
    });
}
const ccBtn = top.createEl("button", { text: "⚙️ Центр", cls: "d3-btn" });
ccBtn.addEventListener("click", () => {
    const pg = dv.pages().where(x => x.file.path.includes("Центр управления")).first();
    if (pg) openNote(pg.file.path);
});
const inBtn = top.createEl("button", { text: "📘 Инструкция", cls: "d3-btn d3-btn-ghost" });
inBtn.addEventListener("click", () => openNote(BASE + "/Инструкция.md"));

/* ── Данные ─────────────────────────────────────────────────────── */
const allNotes = dv.pages('"' + BASE + '"').where(p => p.log && KINDS[String(p.log)] && !p.шаблон);

/* ── Статистика ─────────────────────────────────────────────────── */
const stats = root.createEl("div", { cls: "d3-stats" });
function renderStats() {
    stats.empty();
    for (const k of ORDER) {
        const n = allNotes.where(p => String(p.log) === k).length;
        const s = stats.createEl("div", { cls: "d3-stat" });
        s.createEl("div", { text: String(n), cls: "d3-stat-val" });
        s.createEl("div", { text: KINDS[k].icon + " " + KINDS[k].plural, cls: "d3-stat-lbl" });
    }
}

/* ── Поиск и сортировка ─────────────────────────────────────────── */
const toolRow = root.createEl("div", { cls: "d3-toolrow" });
const searchInput = toolRow.createEl("input", { cls: "d3-input" });
searchInput.type = "text";
searchInput.placeholder = "🔍 Поиск по названию и описанию…";
const sortSel = toolRow.createEl("select", { cls: "d3-select" });
for (const [v, t] of [["new", "Сначала новые"], ["old", "Сначала старые"], ["az", "По алфавиту"], ["type", "По типу"]]) {
    const o = sortSel.createEl("option", { text: t }); o.value = v;
}
let activeSort = "new";
sortSel.addEventListener("change", () => { activeSort = sortSel.value; renderGrid(); });

/* ── Фильтры ────────────────────────────────────────────────────── */
const typeFilters = root.createEl("div", { cls: "d3-filters" });
const activeTypes = new Set();
for (const k of ORDER) {
    const c = typeFilters.createEl("button", { cls: "d3-chip", text: KINDS[k].icon + " " + KINDS[k].plural });
    c.addEventListener("click", () => {
        if (activeTypes.has(k)) activeTypes.delete(k); else activeTypes.add(k);
        c.classList.toggle("is-active", activeTypes.has(k));
        renderGrid();
    });
}
const tagFilters = root.createEl("div", { cls: "d3-filters" });
const activeTags = new Set();
function allTags() {
    const m = new Map();
    for (const p of allNotes.array()) for (const t of toArr(p.теги)) if (t) m.set(t, (m.get(t) || 0) + 1);
    return [...m.entries()].sort((a, b) => b[1] - a[1]).slice(0, 14);
}
function renderTagFilters() {
    tagFilters.empty();
    const tags = allTags();
    if (tags.length === 0) return;
    for (const [t] of tags) {
        const c = tagFilters.createEl("button", { cls: "d3-tagchip", text: "#" + t });
        c.classList.toggle("is-active", activeTags.has(t));
        c.addEventListener("click", () => {
            if (activeTags.has(t)) activeTags.delete(t); else activeTags.add(t);
            renderTagFilters();
            renderGrid();
        });
    }
}

/* ── Сетка ──────────────────────────────────────────────────────── */
const grid = root.createEl("div", { cls: "d3-grid" });

function renderGrid() {
    grid.empty();
    const q = searchInput.value.trim().toLowerCase();
    let items = allNotes.array();
    if (activeTypes.size) items = items.filter(p => activeTypes.has(String(p.log)));
    if (activeTags.size) items = items.filter(p => toArr(p.теги).some(t => activeTags.has(t)));
    if (q) items = items.filter(p => (String(p.title || p.file.name) + " " + String(p.описание || "")).toLowerCase().includes(q));

    const bySort = {
        new: (a, b) => (b.file.mtime?.ts || 0) - (a.file.mtime?.ts || 0),
        old: (a, b) => (a.file.mtime?.ts || 0) - (b.file.mtime?.ts || 0),
        az:  (a, b) => String(a.title || a.file.name).localeCompare(String(b.title || b.file.name), "ru"),
        type:(a, b) => ORDER.indexOf(String(a.log)) - ORDER.indexOf(String(b.log)) || String(a.title || "").localeCompare(String(b.title || ""), "ru"),
    };
    items.sort(bySort[activeSort]);

    if (items.length === 0) {
        grid.createEl("div", { text: "Ничего не найдено. Добавь запись кнопкой «＋» вверху.", cls: "d3-empty" });
        return;
    }
    for (const p of items) {
        const K = KINDS[String(p.log)];
        const col = COLORS[String(p.log)] || "#8a8a8a";
        const card = grid.createEl("div", { cls: "d3-card" });
        const cov = String(p.обложка || p.превью || "");
        const url = cov ? coverUrl(cov, p.file.path) : null;
        if (url) {
            const im = card.createEl("img", { cls: "d3-cover" });
            im.src = url;
        } else {
            card.createEl("div", { cls: "d3-cover-ph", text: K.icon });
        }
        const body = card.createEl("div", { cls: "d3-card-body" });
        const head = body.createEl("div", { cls: "d3-card-head" });
        head.createEl("div", { cls: "d3-card-title", text: String(p.title || p.file.name) });
        const bdg = head.createEl("span", { cls: "d3-badge", text: K.icon });
        bdg.style.cssText = "background:" + col + "22;color:" + col + ";border-color:" + col + ";";
        if (p.описание) body.createEl("div", { cls: "d3-card-desc", text: String(p.описание) });
        if (p.статус) {
            const st = body.createEl("div", { cls: "d3-tags" });
            const sb = st.createEl("span", { cls: "d3-tag", text: "● " + String(p.статус) });
            sb.style.color = col;
        }
        const tags = toArr(p.теги).filter(Boolean);
        if (tags.length) {
            const tw = body.createEl("div", { cls: "d3-tags" });
            for (const t of tags.slice(0, 4)) tw.createEl("span", { cls: "d3-tag", text: "#" + t });
        }
        card.addEventListener("click", () => openNote(p.file.path));
    }
}

/* ── Модалка создания ───────────────────────────────────────────── */
function wkForm(kind) {
    const K = KINDS[kind];
    const ov = document.createElement("div");
    ov.className = "d3-modal";
    const box = ov.createEl("div", { cls: "d3-modal-box" });
    box.createEl("div", { text: K.icon + " Новая запись · " + K.label, cls: "d3-modal-title" });

    const fTitle = box.createEl("div", { cls: "d3-field" });
    fTitle.createEl("label", { text: "Название *" });
    const tIn = fTitle.createEl("input", { cls: "d3-input" });
    tIn.type = "text"; tIn.placeholder = "Например: Мокрый асфальт"; tIn.dataset.f = "name";
    tIn.addEventListener("keydown", e => { if (e.key === "Enter") ok.click(); });

    const fDesc = box.createEl("div", { cls: "d3-field" });
    fDesc.createEl("label", { text: "Описание" });
    const dIn = fDesc.createEl("textarea", { cls: "d3-input" });
    dIn.rows = 2; dIn.placeholder = "Коротко, в одну-две строки"; dIn.dataset.f = "desc";

    const r2 = box.createEl("div", { cls: "d3-row2" });
    const fTags = r2.createEl("div", { cls: "d3-field" });
    fTags.createEl("label", { text: "Теги (через запятую)" });
    const gIn = fTags.createEl("input", { cls: "d3-input" });
    gIn.type = "text"; gIn.placeholder = "lighting, blender"; gIn.dataset.f = "tags";

    const fProj = r2.createEl("div", { cls: "d3-field" });
    fProj.createEl("label", { text: "Проект" });
    const pIn = fProj.createEl("input", { cls: "d3-input" });
    pIn.type = "text"; pIn.placeholder = "необязательно"; pIn.dataset.f = "proj";

    box.createEl("div", { text: "Файл создаётся из шаблона и сразу открывается.", cls: "d3-progress-lbl" });

    const btns = box.createEl("div", { cls: "d3-modal-btns" });
    const cancel = btns.createEl("button", { text: "Отмена", cls: "d3-btn" });
    cancel.dataset.x = "cancel";
    const ok = btns.createEl("button", { text: "Создать", cls: "d3-btn d3-btn-primary" });
    ok.dataset.x = "ok";

    const close = () => ov.remove();
    cancel.addEventListener("click", close);
    ov.addEventListener("click", e => { if (e.target === ov) close(); });
    document.addEventListener("keydown", function esc(e) {
        if (e.key === "Escape") { close(); document.removeEventListener("keydown", esc, true); }
    }, true);

    ok.addEventListener("click", async () => {
        const name = tIn.value.trim();
        if (!name) { tIn.focus(); return; }
        const dir = BASE + "/" + K.dir;
        const newPath = dir + "/" + name + ".md";
        if (app.vault.getAbstractFileByPath(newPath)) {
            dIn.value = "Уже существует: " + name;
            return;
        }
        try {
            const tplPath = dir + "/" + K.tpl;
            const raw = await app.vault.adapter.read(tplPath);
            let content = raw.replace(/\{\{TITLE\}\}/g, name);
            const nf = await app.vault.create(newPath, content);
            const tags = gIn.value.split(",").map(s => s.trim()).filter(Boolean);
            const desc = dIn.value.trim();
            const proj = pIn.value.trim();
            await app.fileManager.processFrontMatter(nf, fm => {
                fm.title = name;
                if (desc) fm.описание = desc;
                if (tags.length) fm.теги = tags;
                if (proj) fm.проект = proj;
                delete fm.шаблон;
            });
            close();
            openNote(newPath);
        } catch (e) {
            dIn.value = "Ошибка: " + (e && e.message || e);
        }
    });

    document.body.appendChild(ov);
    tIn.focus();
}

/* ── Первый рендер ──────────────────────────────────────────────── */
renderStats();
renderTagFilters();
renderGrid();

searchInput.addEventListener("input", renderGrid);
```

> **Как пользоваться:** кнопками «＋» вверху создаются записи нужного типа. Свяжи записи с проектом — укажи поле `проект` в карточке-модалке или прямо в заметке. Все связанные записи соберутся в карточке проекта.
