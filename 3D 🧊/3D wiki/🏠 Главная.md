// Безопасное преобразование любого значения в JS-массив (в т.ч. Dataview-массивы)
function toArr(v) {
    if (v == null) return [];
    if (Array.isArray(v)) return v.map(x => String(x));
    if (typeof v === "object" && typeof v.array === "function") return v.array().map(x => String(x));
    return [String(v)];
}

const allNotes = dv.pages().where(p => p.log && CATS[p.log]);
const root = dv.el("div", "", {});
root.style.cssText = `color: ${C.text}; font-family: sans-serif;`;

// Статистика
const sg = root.createEl("div", { cls: "wk-stats" });
for (const [key, cat] of Object.entries(CATS)) {
    const count = allNotes.filter(p => p.log === key).length;
    const s = sg.createEl("div", { cls: "wk-stat" });
    s.createEl("div", { text: cat.icon + " " + count, cls: "wk-stat-val" });
    s.createEl("div", { text: cat.label, cls: "wk-stat-lbl" });
}

// Поиск
const searchInput = root.createEl("input", { cls: "wk-search" });
searchInput.placeholder = "🔍 Поиск по заметкам...";
searchInput.type = "text";

// Фильтр по категории
let activeCat = "все";
let searchQuery = "";
const filterBar = root.createEl("div", { cls: "wk-filters" });
const catBtns = {};
const allBtn = filterBar.createEl("button", { text: "Все", cls: "wk-btn active" });
catBtns["все"] = allBtn;
allBtn.addEventListener("click", () => { activeCat = "все"; updateBtns(); renderGrid(); });
for (const [key, cat] of Object.entries(CATS)) {
    const btn = filterBar.createEl("button", { text: cat.icon + " " + cat.label, cls: "wk-btn" });
    catBtns[key] = btn;
    btn.addEventListener("click", () => { activeCat = key; updateBtns(); renderGrid(); });
}
function updateBtns() { Object.entries(catBtns).forEach(([k, b]) => b.classList.toggle("active", k === activeCat)); }

// Теги — исправлено: .array() перед flatMap, toArr() для совместимости
const allTags = [...new Set(
    allNotes.array().flatMap(p => toArr(p.теги)).filter(Boolean)
)].sort();
let activeTag = "все";
if (allTags.length > 0) {
    const tagRow = root.createEl("div", { cls: "wk-filters" });
    const allTagBtn = tagRow.createEl("button", { text: "Все теги", cls: "wk-tag-btn active" });
    const tagBtns = { "все": allTagBtn };
    allTagBtn.addEventListener("click", () => {
        activeTag = "все";
        Object.entries(tagBtns).forEach(([k, b]) => b.classList.toggle("active", k === "все"));
        renderGrid();
    });
    for (const t of allTags) {
        const btn = tagRow.createEl("button", { text: t, cls: "wk-tag-btn" });
        tagBtns[t] = btn;
        btn.addEventListener("click", () => {
            activeTag = t;
            Object.entries(tagBtns).forEach(([k, b]) => b.classList.toggle("active", k === t));
            renderGrid();
        });
    }
}

const gridEl = root.createEl("div", { cls: "wk-grid" });

const BADGE_COLORS = {
    "идея":       { bg: "#3d0000", color: C.accent,  border: C.accent },
    "фишка":      { bg: "#1a2a3a", color: "#4a90d9", border: "#4a90d9" },
    "инструмент": { bg: "#1a2a1a", color: C.green,   border: C.green },
    "референс":   { bg: "#2a2a00", color: C.yellow,  border: "#8a8000" },
};

function renderGrid() {
    gridEl.innerHTML = "";
    let list = allNotes.array();
    if (activeCat !== "все") list = list.filter(p => p.log === activeCat);
    if (activeTag !== "все") list = list.filter(p => toArr(p.теги).includes(activeTag));
    if (searchQuery) {
        const q = searchQuery.toLowerCase();
        list = list.filter(p =>
            (p.title || p.file.name).toLowerCase().includes(q) ||
            String(p.описание || "").toLowerCase().includes(q)
        );
    }
    list.sort((a, b) => b.file.mtime - a.file.mtime);

    if (list.length === 0) {
        gridEl.createEl("div", { text: "Ничего не найдено", cls: "wk-empty" });
        return;
    }

    for (const p of list) {
        const cat = CATS[p.log] || { icon: "📄", label: String(p.log) };
        const bc = BADGE_COLORS[p.log] || { bg: C.empty, color: C.muted, border: C.border };
        const card = gridEl.createEl("div", { cls: "wk-card" });
        card.addEventListener("click", () => app.workspace.openLinkText(p.file.path, "", false));

        const head = card.createEl("div", { cls: "wk-card-head" });
        head.createEl("div", { text: cat.icon + " " + (p.title || p.file.name), cls: "wk-card-title" });
        const badge = head.createEl("span", { text: cat.label, cls: "wk-badge" });
        badge.style.cssText = `background: ${bc.bg}; color: ${bc.color}; border-color: ${bc.border};`;

        if (p.описание) card.createEl("div", { text: String(p.описание), cls: "wk-card-desc" });

        const tags = toArr(p.теги);
        if (tags.length > 0) {
            const tagWrap = card.createEl("div", { cls: "wk-tags" });
            for (const t of tags) tagWrap.createEl("span", { text: t, cls: "wk-tag" });
        }

        card.createEl("div", { text: p.file.mtime.toFormat("dd.MM.yyyy"), cls: "wk-date" });
    }
}

searchInput.addEventListener("input", () => { searchQuery = searchInput.value; renderGrid(); });
renderGrid();
