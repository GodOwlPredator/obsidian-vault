# ⚡ Название фишки

```dataviewjs
const C = { accent:"#8f0000",accentLight:"#b30000",accentDim:"#3d0000",card:"#1e1e1e",border:"#2e2e2e",text:"#e0e0e0",muted:"#777",empty:"#2a2a2a",emptyBorder:"#444",green:"#2d7a2d",yellow:"#6a6000" };

function toArr(v) {
    if (v == null) return [];
    if (Array.isArray(v)) return v.map(x => String(x));
    if (typeof v === "object" && typeof v.array === "function") return v.array().map(x => String(x));
    return [String(v)];
}

const p = dv.current(), file = dv.current().file;

async function setValue(key, val) {
    const tf = app.vault.getAbstractFileByPath(file.path);
    const content = await app.vault.read(tf);
    const regex = new RegExp(`^(${key}:\\s*).*$`, "m");
    const updated = regex.test(content)
        ? content.replace(regex, `$1${val}`)
        : content.startsWith("---")
            ? content.replace(/^---\n/, `---\n${key}: ${val}\n`)
            : `---\n${key}: ${val}\n---\n` + content;
    await app.vault.modify(tf, updated);
}

const root = dv.el("div", "", {});
root.style.cssText = `font-family:sans-serif;color:${C.text};`;
const card = root.createEl("div", {});
card.style.cssText = `background:${C.card};border:1px solid ${C.border};border-left:4px solid #4a90d9;border-radius:10px;padding:16px 20px;margin-bottom:12px;`;

card.createEl("h2", { text: "⚡ " + (p.title || file.name) }).style.cssText = `margin:0 0 12px;font-size:18px;color:${C.text};`;

// Сложность
const DIFFS = ["простая", "средняя", "сложная"];
const DIFF_C = { "простая": C.green, "средняя": C.yellow, "сложная": C.accent };
const dRow = card.createEl("div", {});
dRow.style.cssText = "display:flex;gap:5px;flex-wrap:wrap;margin-bottom:10px;align-items:center;";
dRow.createEl("span", { text: "Сложность:" }).style.cssText = `font-size:11px;color:${C.muted};margin-right:2px;`;
for (const d of DIFFS) {
    const active = String(p.сложность) === d;
    const btn = dRow.createEl("button", { text: d });
    btn.style.cssText = `padding:3px 10px;border-radius:5px;border:1px solid ${active ? DIFF_C[d] : C.emptyBorder};background:${active ? DIFF_C[d] : C.empty};color:${active ? "#fff" : C.muted};font-size:11px;font-weight:${active ? "700" : "400"};cursor:pointer;transition: background-color .15s, border-color .15s, color .15s;`;
    btn.addEventListener("mouseenter", () => { if (!active) { btn.style.borderColor = DIFF_C[d]; btn.style.color = C.text; } });
    btn.addEventListener("mouseleave", () => { if (!active) { btn.style.borderColor = C.emptyBorder; btn.style.color = C.muted; } });
    btn.addEventListener("click", async () => { await setValue("сложность", d); });
}

// Программа
if (p.программа) {
    const pr = card.createEl("div", {});
    pr.style.cssText = "display:flex;gap:6px;align-items:center;margin-bottom:10px;";
    pr.createEl("span", { text: "🔧 Программа:" }).style.cssText = `font-size:11px;color:${C.muted};`;
    pr.createEl("span", { text: String(p.программа) }).style.cssText = `background:${C.empty};color:${C.muted};border-radius:5px;padding:3px 9px;font-size:11px;border:1px solid ${C.emptyBorder};`;
}

// Описание
if (p.описание) {
    const d = card.createEl("div", { text: String(p.описание) });
    d.style.cssText = `font-size:13px;color:${C.text};line-height:1.5;padding-top:10px;border-top:1px solid ${C.border};margin-bottom:10px;`;
}

// Теги
const tags = toArr(p.теги).filter(Boolean);
if (tags.length > 0) {
    const tw = card.createEl("div", {});
    tw.style.cssText = "display:flex;gap:5px;flex-wrap:wrap;margin-bottom:8px;";
    for (const t of tags) {
        tw.createEl("span", { text: t }).style.cssText = `background:${C.empty};color:${C.muted};border-radius:4px;padding:2px 8px;font-size:11px;border:1px solid ${C.emptyBorder};`;
    }
}

// Назад
const back = card.createEl("a", { text: "← 3D Вики" });
back.style.cssText = `display:inline-block;font-size:11px;color:${C.accent};text-decoration:none;padding:2px 8px;border:1px solid ${C.accent};border-radius:4px;`;
back.addEventListener("click", (e) => {
    e.preventDefault();
    const home = dv.pages().where(f => f.cssclasses && toArr(f.cssclasses).includes("wiki-3d-home")).first();
    if (home) app.workspace.openLinkText(home.file.path, "", false);
});
```
