# 🧭 Пайплайн производства

```dataviewjs
/* ═══ Пайплайн: этапы 3D-производства + чеклисты ═══ */
async function ensureD3Css() {
    if (document.getElementById("d3-css")) return;
    let loaded = "";
    try { loaded = getComputedStyle(document.documentElement).getPropertyValue("--d3-loaded").trim(); } catch (e) {}
    if (loaded === "1") return;
    let css = "";
    try { css = await app.vault.adapter.read(".obsidian/snippets/3d.css"); } catch (e) {}
    const s = document.createElement("style");
    s.id = "d3-css"; s.textContent = css; document.head.appendChild(s);
}
await ensureD3Css();

function toArr(v) {
    if (v == null) return [];
    if (Array.isArray(v)) return v.map(x => String(x));
    if (typeof v === "object" && typeof v.array === "function") return v.array().map(x => String(x));
    return [String(v)];
}
function findPage(cls) {
    return dv.pages().where(x => x.cssclasses && toArr(x.cssclasses).includes(cls)).first();
}
function openNote(p) { app.workspace.openLinkText(p, "", false); }

const root = dv.el("div", "", {});
root.className = "d3-root";

const top = root.createEl("div", { cls: "d3-top" });
const home = findPage("wiki-3d-home");
if (home) { const b = top.createEl("button", { text: "← 3D Вики", cls: "d3-btn" }); b.addEventListener("click", () => openNote(home.file.path)); }
const gal = findPage("gallery-home");
if (gal) { const b = top.createEl("button", { text: "🖼 Галерея", cls: "d3-btn" }); b.addEventListener("click", () => openNote(gal.file.path)); }

/* ── Этапы производства ─────────────────────────────────────────── */
const STAGES = [
    ["🧱", "Blockout", "Примари-формы: силуэт, пропорции, композиция, масштаб. Никаких деталей — только большие массы."],
    ["🗿", "High-poly", "Детализация и лепка. Сколы, швы, орнамент — всё, что читается в силуэте."],
    ["🔁", "Retopo", "Чистая топология: квады, петли под деформацию, адекватная плотность под сглаживание."],
    ["📐", "UV", "Развёртки без перекрытий и растяжений. Texel density одинаковый, паковка плотная."],
    ["🧪", "Bake", "Нормали, AO, curvature, thickness, displacement. Проверь артефакты по швам."],
    ["🎨", "Texturing", "PBR: albedo / roughness / metalness / height. Пятна, крупные формы, детали, цветовые вариации."],
    ["💡", "Lookdev / Свет", "HDRI + key/fill/rim. Проверь материал на сером, белом и цветном фоне."],
    ["🎥", "Рендер", "Сэмплы, денойз, passes (beauty, AO, depth, cryptomatte). Сохрани всё — пригодится в композе."],
    ["🎞", "Композитинг", "Цветокоррекция, bloom, глубина, грейдинг. Собери финальную картинку."],
    ["🚀", "Подача", "ArtStation / портфолио: превью 16:10, описание, теги, время работы. Занеси в галерею."],
];

root.createEl("div", { text: "Этапы производства", cls: "d3-section" });
const sgrid = root.createEl("div", { cls: "d3-grid" });
STAGES.forEach(([icon, name, hint], i) => {
    const card = sgrid.createEl("div", { cls: "d3-card" });
    card.style.cursor = "default";
    const body = card.createEl("div", { cls: "d3-card-body" });
    const head = body.createEl("div", { cls: "d3-card-head" });
    head.createEl("div", { cls: "d3-card-title", text: icon + " " + (i + 1) + ". " + name });
    body.createEl("div", { cls: "d3-card-desc", text: hint });
});

/* ── Чеклисты ───────────────────────────────────────────────────── */
root.createEl("div", { text: "Чеклисты", cls: "d3-section" });
const lists = dv.pages().where(p => String(p.log) === "чеклист" && !p.шаблон);
const wrap = root.createEl("div", { cls: "d3-grid" });
if (lists.length === 0) {
    root.createEl("div", { text: "Чеклистов пока нет. Создай их на главной: «＋ Чеклист».", cls: "d3-empty" });
}
for (const p of lists.array()) {
    const card = wrap.createEl("div", { cls: "d3-card" });
    const body = card.createEl("div", { cls: "d3-card-body" });
    body.createEl("div", { cls: "d3-card-title", text: "✅ " + String(p.title || p.file.name) });
    if (p.описание) body.createEl("div", { cls: "d3-card-desc", text: String(p.описание) });
    let done = 0, total = 0;
    try {
        const raw = await app.vault.adapter.read(p.file.path);
        for (const line of raw.split("\n")) {
            const m = line.match(/^\s*[-*]\s+\[([ xX])\]/);
            if (m) { total++; if (m[1] !== " ") done++; }
        }
    } catch (e) {}
    const pct = total ? Math.round((done / total) * 100) : 0;
    const bar = body.createEl("div", { cls: "d3-progress" });
    bar.createEl("i", {}).style.width = pct + "%";
    body.createEl("div", { cls: "d3-progress-lbl", text: done + " / " + total + " · " + pct + "%" });
    card.addEventListener("click", () => openNote(p.file.path));
}
```

> Этапы — это каркас, а не догма. Для мелких ассетов можно пропустить композитинг, для крупных — добавить отдельный этап симуляции/гримминга.
