    if (document.getElementById("d3-css")) return;

function toArr(v) {
    if (v == null) return [];
    if (Array.isArray(v)) return v.map(x => String(x));
    if (typeof v === "object" && typeof v.array === "function") return v.array().map(x => String(x));
    return [String(v)];
}

async function createWork() {
    const d = await glForm();
    if (!d) return;
    const safe = d.name.replace(/[\\/:*?"<>|#^\[\]]/g, " ").replace(/\s+/g, " ").trim();
    if (!safe) { new Notice("Некорректное название"); return; }
    const path = WORK_DIR + "/" + safe + ".md";
    if (app.vault.getAbstractFileByPath(path)) { new Notice("Уже существует: " + safe); return; }
    const tf = app.vault.getAbstractFileByPath(WORK_TPL);
    if (!tf) { new Notice("Не найден шаблон: " + WORK_TPL); return; }
    try {
        let raw = await app.vault.read(tf);
        raw = raw.replace(/\{\{TITLE\}\}/g, d.name);
        if (/^#\s+.*$/m.test(raw)) raw = raw.replace(/^#\s+.*$/m, "# 🎨 " + d.name);
        const nf = await app.vault.create(path, raw);
        await app.fileManager.processFrontMatter(nf, fm => {
            fm.log = "работа";
            fm.title = d.name;
            delete fm.шаблон;
            if (d.desc) fm.описание = d.desc;
            if (d.category) fm.категория = d.category;
            if (d.year) fm.год = Number(d.year) || d.year;
            if (d.progs.length) fm.программы = d.progs;
            if (d.tags.length) fm.теги = d.tags;
        });
        await app.workspace.getLeaf(false).openFile(nf);
        new Notice("✅ Работа создана: " + safe);
    } catch (e) { new Notice("Ошибка создания: " + e); }
}

const gridEl = root.createEl("div", { cls: "d3-grid" });
function renderGrid() {
    gridEl.empty();
    let list = worksArr.slice();
    if (activeCat !== "все") list = list.filter(w => String(w.категория || "") === activeCat);
    if (activeYear !== "все") list = list.filter(w => String(w.год || "") === activeYear);
    if (searchQuery) {
        const q = searchQuery.toLowerCase();
        list = list.filter(w => (String(w.title || w.file.name) + " " + String(w.описание || "")).toLowerCase().includes(q));
    }
    list.sort((a, b) => {
        if (activeSort === "rating") return (Number(b.рейтинг) || 0) - (Number(a.рейтинг) || 0);
        if (activeSort === "title") return String(a.title || a.file.name).localeCompare(String(b.title || b.file.name), "ru");
        return (b.file.mtime?.ts || 0) - (a.file.mtime?.ts || 0);
    });
    if (list.length === 0) { gridEl.createEl("div", { text: "Работы не найдены", cls: "d3-empty" }); return; }
    for (const w of list) {
        const card = gridEl.createEl("div", { cls: "d3-card" });

        head.createEl("div", { cls: "d3-card-title", text: String(w.title || w.file.name) });
        if (w.категория) {
            const b = head.createEl("span", { cls: "d3-badge", text: String(w.категория) });
            b.style.cssText = "background:#a8000022;color:#d13b3b;border-color:#a80000;";
        }
        const meta = [];
        if (w.год) meta.push(String(w.год));
        const progs = toArr(w.программы).filter(Boolean);
