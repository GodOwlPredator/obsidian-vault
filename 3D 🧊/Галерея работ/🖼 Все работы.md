function toArr(v) {
    if (v == null) return [];
    if (Array.isArray(v)) return v.map(x => String(x));
    if (typeof v === "object" && typeof v.array === "function") return v.array().map(x => String(x));
    return [String(v)];
}

// ─── Сетка работ ──────────────────────────────────────────────────────────────
const gridEl = root.createEl("div", { cls: "gl-grid" });

function renderGrid() {
    gridEl.innerHTML = "";
    let list = allWorks.array();
    if (activeCat !== "все")  list = list.filter(w => String(w.категория || "") === activeCat);
    if (activeYear !== "все") list = list.filter(w => String(w.год || "") === activeYear);
    if (searchQuery) {
        const q = searchQuery.toLowerCase();
        list = list.filter(w =>
            (w.title || w.file.name).toLowerCase().includes(q) ||
            String(w.описание || "").toLowerCase().includes(q)
        );
    }
    list.sort((a, b) => {
        if (activeSort === "rating") return (Number(b.рейтинг) || 0) - (Number(a.рейтинг) || 0);
        if (activeSort === "title")  return String(a.title || a.file.name).localeCompare(String(b.title || b.file.name));
        return b.file.mtime - a.file.mtime;
    });

    if (list.length === 0) {
        gridEl.createEl("div", { text: "Работы не найдены", cls: "gl-empty" });
        return;
    }

    for (const w of list) {
        const card = gridEl.createEl("div", { cls: "gl-card" });
        card.addEventListener("click", () => app.workspace.openLinkText(w.file.path, "", false));

        const cover = String(w.превью || "");
        if (cover) {
            const cp = cover.replace(/^\[\[|\]\]$/g, "");
            const tf = app.vault.getAbstractFileByPath(cp) || app.metadataCache.getFirstLinkpathDest(cp, w.file.path);
            if (tf) {
                const img = card.createEl("img", { cls: "gl-thumb" });
                img.src = app.vault.getResourcePath(tf);
            } else {
                card.createEl("div", { cls: "gl-thumb-ph" }).textContent = "🎨";
            }
        } else {
            card.createEl("div", { cls: "gl-thumb-ph" }).textContent = "🎨";
        }

        const info = card.createEl("div", { cls: "gl-info" });
        if (w.категория) {
            const badge = info.createEl("span", { text: String(w.категория), cls: "gl-badge" });
            badge.style.cssText = `background:${C.accentDim};color:${C.accentLight};border:1px solid ${C.accent};`;
        }
        info.createEl("div", { text: w.title || w.file.name, cls: "gl-title" });
        const metaLine = [];
        if (w.год) metaLine.push(String(w.год));
        const progs = toArr(w.программы).filter(Boolean);
        if (progs.length > 0) metaLine.push(progs.join(", "));
        if (metaLine.length > 0) info.createEl("div", { text: metaLine.join(" · "), cls: "gl-meta" });
        const rating = Number(w.рейтинг) || 0;
        if (rating > 0) info.createEl("div", { text: "★".repeat(rating) + "☆".repeat(5 - rating), cls: "gl-stars" });
        const tags = toArr(w.теги).filter(Boolean);
        if (tags.length > 0) {
            const tw = info.createEl("div", { cls: "gl-tags" });
            for (const t of tags.slice(0, 3)) tw.createEl("span", { text: t, cls: "gl-tag" });
        }
    }
}
