# 🎨 Название работы

```dataviewjs
const C = { accent:"#8f0000",accentLight:"#b30000",accentDim:"#3d0000",card:"#1e1e1e",border:"#2e2e2e",text:"#e0e0e0",muted:"#777",empty:"#2a2a2a",emptyBorder:"#444",green:"#2d7a2d",yellow:"#6a6000" };

function toArr(v) {
    if (v == null) return [];
    if (Array.isArray(v)) return v.map(x => String(x));
    if (typeof v === "object" && typeof v.array === "function") return v.array().map(x => String(x));
    return [String(v)];
}

const p = dv.current(), file = dv.current().file;
const rating = Number(p.рейтинг) || 0;

async function setValue(key, val) {
    const tf = app.vault.getAbstractFileByPath(file.path);
    const content = await app.vault.read(tf);
    const regex = new RegExp(`^(${key}:\\s*).*$`, "m");
    const updated = regex.test(content)
        ? content.replace(regex, `$1${val}`)
        : content.startsWith("---")
            ? content.replace(/^---\n/, `---\n${key}: ${val}\n`)
            : `---\n${key}: ${val}\n---\n` + content;
    await app.vault.modify(tf, updated);
}

const root = dv.el("div", "", {});
root.style.cssText = `font-family:sans-serif;color:${C.text};`;
const card = root.createEl("div", {});
card.style.cssText = `background:${C.card};border:1px solid ${C.border};border-radius:12px;padding:16px;margin-bottom:12px;display:flex;gap:20px;align-items:flex-start;flex-wrap:wrap;`;

// Превью
const coverWrap = card.createEl("div", {});
coverWrap.style.cssText = "flex-shrink:0;";
const cover = String(p.превью || "");
if (cover) {
    const cp = cover.replace(/^\[\[|\]\]$/g, "");
    const tf2 = app.vault.getAbstractFileByPath(cp) || app.metadataCache.getFirstLinkpathDest(cp, file.path);
    if (tf2) {
        const img = coverWrap.createEl("img", {});
        img.src = app.vault.getResourcePath(tf2);
        img.style.cssText = "width:180px;height:180px;object-fit:cover;border-radius:8px;display:block;";
    } else {
        const ph = coverWrap.createEl("div", {});
        ph.style.cssText = `width:180px;height:180px;background:linear-gradient(135deg,${C.accentDim},#111);border-radius:8px;display:flex;align-items:center;justify-content:center;font-size:50px;`;
        ph.textContent = "🎨";
    }
} else {
    const ph = coverWrap.createEl("div", {});
    ph.style.cssText = `width:180px;height:180px;background:linear-gradient(135deg,${C.accentDim},#111);border-radius:8px;display:flex;align-items:center;justify-content:center;font-size:50px;`;
    ph.textContent = "🎨";
}

// Мета
const meta = card.createEl("div", {});
meta.style.cssText = "flex:1;min-width:200px;display:flex;flex-direction:column;gap:10px;";

meta.createEl("h2", { text: p.title || file.name }).style.cssText = `margin:0 0 3px;font-size:20px;color:${C.text};`;

if (p.описание) {
    meta.createEl("div", { text: String(p.описание) }).style.cssText = `font-size:13px;color:${C.muted};line-height:1.5;`;
}

// Категория, программы, год, время
const tagsRow = meta.createEl("div", {});
tagsRow.style.cssText = "display:flex;gap:6px;flex-wrap:wrap;";
if (p.категория) {
    const t = tagsRow.createEl("span", { text: String(p.категория) });
    t.style.cssText = `background:${C.accentDim};color:${C.accentLight};border-radius:5px;padding:3px 9px;font-size:11px;border:1px solid ${C.accent};font-weight:600;`;
}
const progs = toArr(p.программы).filter(Boolean);
for (const prog of progs) {
    tagsRow.createEl("span", { text: prog }).style.cssText = `background:${C.empty};color:${C.muted};border-radius:5px;padding:3px 9px;font-size:11px;border:1px solid ${C.emptyBorder};`;
}
if (p.год) tagsRow.createEl("span", { text: "📅 " + String(p.год) }).style.cssText = `background:${C.empty};color:${C.muted};border-radius:5px;padding:3px 9px;font-size:11px;border:1px solid ${C.emptyBorder};`;
if (p.время_работы) tagsRow.createEl("span", { text: "⏱ " + String(p.время_работы) }).style.cssText = `background:${C.empty};color:${C.muted};border-radius:5px;padding:3px 9px;font-size:11px;border:1px solid ${C.emptyBorder};`;

// Рейтинг
const rr = meta.createEl("div", {});
rr.style.cssText = "display:flex;align-items:center;gap:4px;";
rr.createEl("span", { text: "Оценка:" }).style.cssText = `font-size:12px;color:${C.muted};margin-right:4px;`;
for (let i = 1; i <= 5; i++) {
    const star = rr.createEl("span", { text: i <= rating ? "★" : "☆" });
    star.style.cssText = `font-size:22px;cursor:pointer;color:${i <= rating ? C.accent : C.muted};transition:color 0.1s,transform 0.1s;`;
    star.addEventListener("mouseenter", () => { star.style.transform = "scale(1.2)"; star.style.color = C.accent; });
    star.addEventListener("mouseleave", () => { star.style.transform = "scale(1)"; star.style.color = i <= rating ? C.accent : C.muted; });
    star.addEventListener("click", async () => { await setValue("рейтинг", i === rating ? 0 : i); });
}

// Избранное + ссылка
const actRow = meta.createEl("div", {});
actRow.style.cssText = "display:flex;gap:8px;align-items:center;flex-wrap:wrap;";
const isFav = p.избранное === true;
const favBtn = actRow.createEl("button", { text: isFav ? "★ В избранном" : "☆ В избранное" });
favBtn.style.cssText = `padding:4px 12px;border-radius:5px;border:1px solid ${isFav ? C.accent : C.emptyBorder};background:${isFav ? C.accentDim : C.empty};color:${isFav ? C.accentLight : C.muted};font-size:11px;cursor:pointer;transition: background-color .15s, border-color .15s, color .15s;`;
favBtn.addEventListener("mouseenter", () => { favBtn.style.borderColor = C.accent; });
favBtn.addEventListener("mouseleave", () => { favBtn.style.borderColor = isFav ? C.accent : C.emptyBorder; });
favBtn.addEventListener("click", async () => { await setValue("избранное", !p.избранное); });

const linkStr = String(p.ссылка || "");
if (linkStr.startsWith("http")) {
    const a = actRow.createEl("a", { text: "🔗 Посмотреть" });
    a.href = linkStr;
    a.target = "_blank";
    a.style.cssText = `font-size:11px;color:${C.accent};text-decoration:none;padding:4px 12px;border:1px solid ${C.accent};border-radius:5px;`;
}

// Теги
const tags = toArr(p.теги).filter(Boolean);
if (tags.length > 0) {
    const tw = meta.createEl("div", {});
    tw.style.cssText = "display:flex;gap:5px;flex-wrap:wrap;";
    for (const t of tags) {
        tw.createEl("span", { text: t }).style.cssText = `background:${C.empty};color:${C.muted};border-radius:4px;padding:2px 8px;font-size:11px;border:1px solid ${C.emptyBorder};`;
    }
}

// Назад
const back = meta.createEl("a", { text: "← Галерея" });
back.style.cssText = `font-size:11px;color:${C.accent};text-decoration:none;padding:2px 8px;border:1px solid ${C.accent};border-radius:4px;align-self:flex-start;`;
back.addEventListener("click", (e) => {
    e.preventDefault();
    const home = dv.pages().where(f => f.cssclasses && toArr(f.cssclasses).includes("gallery-home")).first();
    if (home) app.workspace.openLinkText(home.file.path, "", false);
});
```
