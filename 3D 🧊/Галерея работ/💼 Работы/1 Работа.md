function toArr(v) {
    if (v == null) return [];
    if (Array.isArray(v)) return v.map(x => String(x));
    if (typeof v === "object" && typeof v.array === "function") return v.array().map(x => String(x));
    return [String(v)];
}

function d3BaseOf(cls) {
    const pg = dv.pages().where(x => x.cssclasses && toArr(x.cssclasses).includes(cls)).first();
    return pg ? pg.file.folder : null;
}

async function setValue(key, val) {
    const tf = app.vault.getAbstractFileByPath(file.path);
    const content = await app.vault.read(tf);
    const re = new RegExp("^(" + key + ":\\s*).*$", "m");
    const upd = re.test(content)
        ? content.replace(re, "$1" + val)
        : content.startsWith("---")
            ? content.replace(/^---\n/, "---\n" + key + ": " + val + "\n")
            : "---\n" + key + ": " + val + "\n---\n" + content;
    await app.vault.modify(tf, upd);
}

function coverFile(str) {
    const cp = String(str || "").replace(/^!?\[\[|\]\]$/g, "").trim();
    if (!cp) return null;
    return app.vault.getAbstractFileByPath(cp) || app.metadataCache.getFirstLinkpathDest(cp, file.path);
}

function openLightbox(src) {
    const lb = document.createElement("div");
    lb.className = "d3-lightbox";
    const img = document.createElement("img");
    img.src = src;
    const hint = document.createElement("div");
    hint.className = "d3-lb-hint";
    hint.textContent = "Клик или Esc — закрыть";
    lb.appendChild(img); lb.appendChild(hint);
    document.body.appendChild(lb);
    const close = () => { lb.remove(); document.removeEventListener("keydown", onKey, true); };
    function onKey(e) { if (e.key === "Escape") { e.preventDefault(); close(); } }
    document.addEventListener("keydown", onKey, true);
    lb.addEventListener("click", close);
}
