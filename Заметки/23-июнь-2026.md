const file = dv.current().file;
const energy = dv.current().energy ?? 0;
const mood = dv.current().mood ?? 0;

const FILL_COLOR = "#8f0000";
const FILL_BORDER = "#b30000";
const FILL_HOVER = "#6a0000";
const EMPTY_COLOR = "#2a2a2a";
const EMPTY_BORDER = "#444";
const EMPTY_HOVER = "#3d0000";

async function setValue(frontmatterKey, value) {
    const tf = app.vault.getAbstractFileByPath(file.path);
    const content = await app.vault.read(tf);
    const regex = new RegExp(`^(${frontmatterKey}:\\s*).*$`, "m");
    let newContent;
    if (regex.test(content)) {
        newContent = content.replace(regex, `$1${value}`);
    } else if (content.startsWith("---")) {
        newContent = content.replace(/^---\n/, `---\n${frontmatterKey}: ${value}\n`);
    } else {
        newContent = `---\n${frontmatterKey}: ${value}\n---\n` + content;
    }
    await app.vault.modify(tf, newContent);
}

function createBar(label, currentValue, frontmatterKey) {
    const wrapper = dv.el("div", "", {});
    wrapper.style.cssText = "margin: 14px 0;";

    const labelEl = wrapper.createEl("div", { text: label });
    labelEl.style.cssText =
        "font-size: 13px; font-weight: 600; letter-spacing: 0.04em; text-transform: uppercase; margin-bottom: 6px; opacity: 0.75;";

    const row = wrapper.createEl("div", {});
    row.style.cssText = "display: flex; gap: 4px; align-items: center;";

    for (let i = 1; i <= 10; i++) {
        const seg = row.createEl("div", {});
        const filled = i <= currentValue;

        seg.style.cssText = `
            width: 30px;
            height: 30px;
            border-radius: 5px;
            cursor: pointer;
            background: ${filled ? FILL_COLOR : EMPTY_COLOR};
            border: 1px solid ${filled ? FILL_BORDER : EMPTY_BORDER};
            display: flex;
            align-items: center;
            justify-content: center;
            font-size: 11px;
            font-weight: ${filled ? "700" : "400"};
            color: ${filled ? "#fff" : "#666"};
            transition: background 0.12s ease, transform 0.1s ease, box-shadow 0.12s ease;
            user-select: none;
        `;
        seg.textContent = i;

        seg.addEventListener("mouseenter", () => {
            seg.style.background = filled ? FILL_HOVER : EMPTY_HOVER;
            seg.style.transform = "scale(1.1)";
            seg.style.boxShadow = `0 0 8px ${FILL_COLOR}88`;
        });
        seg.addEventListener("mouseleave", () => {
            seg.style.background = filled ? FILL_COLOR : EMPTY_COLOR;
            seg.style.transform = "scale(1)";
            seg.style.boxShadow = "none";
        });

        seg.addEventListener("click", async () => {
            const newVal = (i === currentValue) ? 0 : i;
            await setValue(frontmatterKey, newVal);
        });
    }

    const valueLabel = row.createEl("span", {});
    valueLabel.style.cssText = `
        margin-left: 10px;
        font-size: 16px;
        font-weight: 700;
        color: ${currentValue > 0 ? FILL_COLOR : "#555"};
        min-width: 40px;
    `;
    valueLabel.textContent = currentValue > 0 ? `${currentValue}/10` : "—";

    return wrapper;
}

const container = dv.el("div", "", {});
container.style.cssText = "padding: 12px 0;";

createBar("Энергия", energy, "energy");
createBar("Настроение", mood, "mood");
