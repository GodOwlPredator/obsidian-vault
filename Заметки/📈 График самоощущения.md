const C = {

    accentLight: "#b30000",

const SVG_NS = "http://www.w3.org/2000/svg";
function svgEl(tag, attrs = {}) {
    const el = document.createElementNS(SVG_NS, tag);
    for (const [k, v] of Object.entries(attrs)) el.setAttribute(k, String(v));
    return el;
}

// ─── Парсинг дат из имён файлов ("13-май-2026" и "13-май.-2026") ─────────────
const MONTHS_RU = {
    "январь":0,"февраль":1,"март":2,"апрель":3,"май":4,"июнь":5,
    "июль":6,"август":7,"сентябрь":8,"октябрь":9,"ноябрь":10,"декабрь":11,
    "янв":0,"фев":1,"февр":1,"мар":2,"март":2,"апр":3,"июн":5,"июл":6,"авг":7,"сен":8,"сент":8,"окт":9,"ноя":10,"нояб":10,"дек":11
};
const MONTHS_FULL = ["Январь","Февраль","Март","Апрель","Май","Июнь",
                     "Июль","Август","Сентябрь","Октябрь","Ноябрь","Декабрь"];

function parseRuDate(name) {
    // Точка после месяца необязательна: оба варианта считаются одной датой.
    const m = name.match(/^(\d{1,2})-([а-яёА-ЯЁ]+)\.?-(\d{4})$/);
    if (!m) return null;
    const month = MONTHS_RU[m[2].toLowerCase()];
    if (month === undefined) return null;
    return new Date(parseInt(m[3]), month, parseInt(m[1]));
}

// ─── Данные ───────────────────────────────────────────────────────────────────
const allEntries = dv.pages()
    .where(p => p.energy !== undefined || p.mood !== undefined)
    .array()
    .map(p => {
        const date = parseRuDate(p.file.name);
        return { page: p, date, energy: Number(p.energy)||0, mood: Number(p.mood)||0, path: p.file.path, name: p.file.name };
    })
    .filter(e => e.date)
    .sort((a, b) => a.date - b.date);

// Годы берём из данных, а месяцы оставляем все 12:
// пропущенный месяц должен быть доступен для выбора и показывать пустой график.
const availableYears = [...new Set(allEntries.map(e => e.date.getFullYear()))].sort((a,b) => a-b);
const latestEntry = allEntries[allEntries.length - 1];
let activeYear = latestEntry ? latestEntry.date.getFullYear() : new Date().getFullYear();
let activeMonth = latestEntry ? latestEntry.date.getMonth() : new Date().getMonth();

// ─── Стили ────────────────────────────────────────────────────────────────────
let st = document.getElementById("mood-chart-style");
if (!st) {
    st = document.createElement("style");
    st.id = "mood-chart-style";
    document.head.appendChild(st);
}
st.textContent = `
        .mc-btn { padding: 4px 14px; border-radius: 16px; border: 1px solid #2e2e2e; background: #1e1e1e; color: #777; font-size: 12px; cursor: pointer; transition: background-color .15s, border-color .15s, color .15s; }
        .mc-btn.active { background: #8f0000; border-color: #b30000; color: #fff; }
        .mc-btn:hover:not(.active) { border-color: #8f0000; color: #e0e0e0; }
        .mc-year-row { display:flex; gap:8px; flex-wrap:wrap; align-items:center; margin-bottom:10px; position:relative; z-index:3; }
        .mc-month-row { display:flex; gap:8px; flex-wrap:wrap; align-items:flex-start; margin-bottom:16px; }
        .mc-month-buttons { display:flex; gap:8px; flex-wrap:wrap; align-items:center; flex:1; min-width:0; }
        .mc-year-control { position:relative; display:inline-block; }
        .mc-year-button { min-width:66px; text-align:center; }
        .mc-year-menu { position:static; min-width:100%; max-height:180px; overflow-y:auto; margin-top:5px; padding:4px; border:1px solid ${C.border}; border-radius:16px; background:#1e1e1e; box-shadow:0 6px 16px rgba(0,0,0,0.45); display:none; }
        .mc-year-menu.open { display:block; }
        .mc-year-option { display:block; width:100%; padding:5px 10px; border:0; border-radius:12px; background:transparent; color:${C.muted}; font-size:12px; text-align:center; cursor:pointer; }
        .mc-year-option:hover, .mc-year-option.active { background:${C.accentDim}; color:#fff; }
        .mc-period { font-size:11px; color:${C.muted}; white-space:nowrap; }
        .mc-entry-count { margin-left:auto; font-size:11px; color:${C.muted}; white-space:nowrap; }
        .mc-tip { position: fixed; background: #111; border: 1px solid #b30000; color: #e0e0e0; font-size: 12px; padding: 7px 11px; border-radius: 7px; pointer-events: none; z-index: 9999; display: none; line-height: 1.7; box-shadow: 0 4px 16px rgba(143,0,0,0.3); }
        .mc-dot { cursor: pointer; transition: r 0.1s; }
        .mc-dot:hover { r: 6; }
`;

// Тултип (один на всю страницу, переиспользуем)
let tip = document.getElementById("mc-global-tip");
if (!tip) { tip = document.createElement("div"); tip.id = "mc-global-tip"; tip.className = "mc-tip"; document.body.appendChild(tip); }

// ─── Корневой контейнер ───────────────────────────────────────────────────────
const root = dv.el("div","",{});
root.style.cssText = `font-family: sans-serif; color: ${C.text};`;

// Фильтр: год сверху, месяцы ниже — как в первоначальной версии.
const yearRow = root.createEl("div",{cls:"mc-year-row"});
const yearControl = yearRow.createEl("div",{cls:"mc-year-control"});
const yearButton = yearControl.createEl("button",{text:String(activeYear),cls:"mc-btn mc-year-button",attr:{"aria-label":"Выбрать год","aria-expanded":"false"}});
const yearMenu = yearControl.createEl("div",{cls:"mc-year-menu"});
const yearOptions = {};
for (const year of (availableYears.length ? availableYears : [activeYear])) {
    const option = yearMenu.createEl("button",{text:String(year),cls:"mc-year-option"+(year === activeYear ? " active" : ""),attr:{type:"button"}});
    yearOptions[year] = option;
    option.addEventListener("click",()=>{
        activeYear = year;
        Object.entries(yearOptions).forEach(([key, button]) =>
            button.classList.toggle("active", Number(key) === activeYear)
        );
        yearButton.textContent = String(year);
        yearButton.setAttribute("aria-expanded","false");
        yearMenu.classList.remove("open");
        renderMonthButtons();
        redraw();
    });
}
yearButton.addEventListener("click",(event)=>{
    event.stopPropagation();
    const isOpen = yearMenu.classList.toggle("open");
    yearButton.setAttribute("aria-expanded",String(isOpen));
});
root.addEventListener("click",(event)=>{
    if (!yearControl.contains(event.target)) {
        yearMenu.classList.remove("open");
        yearButton.setAttribute("aria-expanded","false");
    }
});

const monthRow = root.createEl("div",{cls:"mc-month-row"});
const monthButtonsWrap = monthRow.createEl("div",{cls:"mc-month-buttons"});
let monthButtons = {};

function renderMonthButtons() {
    monthButtonsWrap.innerHTML = "";
    monthButtons = {};
    const availableForYear = [...new Set(
        allEntries
            .filter(e => e.date.getFullYear() === activeYear)
            .map(e => e.date.getMonth())
    )].sort((a,b) => a-b);

    if (!availableForYear.includes(activeMonth)) {
        activeMonth = availableForYear.length ? availableForYear[availableForYear.length - 1] : null;
    }

    for (const month of availableForYear) {
        const btn = monthButtonsWrap.createEl("button",{text:MONTHS_FULL[month],cls:"mc-btn"});
        monthButtons[month] = btn;
        btn.addEventListener("click",()=>{
            activeMonth = month;
            Object.entries(monthButtons).forEach(([key, button]) =>
                button.classList.toggle("active", Number(key) === activeMonth)
            );
            redraw();
        });
    }

    if (!availableForYear.length) {
        monthButtonsWrap.createEl("span",{text:"Нет записей за этот год",cls:"mc-period"});
    } else {
        Object.entries(monthButtons).forEach(([key, button]) =>
            button.classList.toggle("active", Number(key) === activeMonth)
        );
    }
}

renderMonthButtons();

// Легенда
const legend = root.createEl("div",{});
legend.style.cssText = `display:flex;gap:18px;margin-bottom:10px;padding:0 2px 9px;font-size:12px;color:${C.muted};border-bottom:1px solid ${C.border};`;
function makeLeg(color,label) {
    const d = legend.createEl("div",{});
    d.style.cssText = "display:flex;align-items:center;gap:6px;";
    const dot = d.createEl("span",{});
    dot.style.cssText = `width:10px;height:10px;border-radius:50%;background:${color};flex-shrink:0;`;

}
const moodLeg = makeLeg(C.accent,"Настроение");
const energyLeg = makeLeg(C.energy,"Энергия");
const entryCount = legend.createEl("span",{cls:"mc-entry-count"});

// Обёртка графика
const chartWrap = root.createEl("div",{});
chartWrap.style.cssText = `background:${C.card};border:1px solid ${C.border};border-radius:10px;padding:14px;overflow:hidden;`;

// Пустой экран
const emptyMsg = chartWrap.createEl("div",{});
emptyMsg.style.cssText = `color:${C.muted};text-align:center;padding:60px 0;font-size:14px;display:none;`;

// SVG-контейнер
const svgWrap = chartWrap.createEl("div",{});
svgWrap.style.cssText = "position:relative;";

// ─── Отрисовка ────────────────────────────────────────────────────────────────
const W = 680, H = 280;
const PAD = { top: 20, right: 24, bottom: 44, left: 34 };
const chartW = W - PAD.left - PAD.right;
const chartH = H - PAD.top - PAD.bottom;

function avg(arr) { return arr.length ? (arr.reduce((a,b)=>a+b,0)/arr.length).toFixed(1) : "—"; }

function redraw() {
    svgWrap.innerHTML = "";
    emptyMsg.style.display = "none";

    const filtered = allEntries.filter(e =>
        e.date.getFullYear() === activeYear && e.date.getMonth() === activeMonth
    );
    const periodLabel = activeMonth === null
        ? `Нет записей за ${activeYear}`
        : `${MONTHS_FULL[activeMonth]} ${activeYear}`;
    entryCount.textContent = `Записей: ${filtered.filter(e => e.energy > 0 || e.mood > 0).length}`;
    emptyMsg.textContent = activeMonth === null
        ? periodLabel
        : `Нет записей за ${periodLabel}`;

    if (filtered.length === 0) { emptyMsg.style.display = "block"; return; }
    emptyMsg.style.display = "none";

    // Обновляем легенду
    const moods = filtered.map(e=>e.mood).filter(v=>v>0);
    const energies = filtered.map(e=>e.energy).filter(v=>v>0);
    moodLeg.querySelector("span:last-child").textContent = `Настроение (ср. ${avg(moods)})`;
    energyLeg.querySelector("span:last-child").textContent = `Энергия (ср. ${avg(energies)})`;

    const n = filtered.length;
    const xStep = n > 1 ? chartW / (n-1) : chartW;

    function xPos(i)  { return PAD.left + (n>1 ? i*xStep : chartW/2); }
    function yPos(v)  { return PAD.top + chartH - (v/10)*chartH; }

    const svg = svgEl("svg", { viewBox:`0 0 ${W} ${H}`, width:"100%", style:"display:block;overflow:visible;" });
    svgWrap.appendChild(svg);

    // Фон
    svg.appendChild(svgEl("rect",{ x:PAD.left,y:PAD.top,width:chartW,height:chartH,fill:"#161616",rx:6 }));

    // Горизонтальные линии сетки (0,2,4,6,8,10)
    for (let v=0;v<=10;v+=2) {
        const y = yPos(v);
        const line = svgEl("line",{ x1:PAD.left,y1:y,x2:PAD.left+chartW,y2:y,stroke:C.grid,"stroke-width":"1" });
        svg.appendChild(line);
        const lbl = svgEl("text",{ x:PAD.left-6,y:y+4,"text-anchor":"end","font-size":"10","fill":C.muted });
        lbl.textContent = v;
        svg.appendChild(lbl);
    }

    // X-метки: показываем каждые N дней, чтобы не перекрываться
    const maxLabels = 12;
    const step = Math.max(1, Math.ceil(n/maxLabels));
    for (let i=0;i<n;i+=step) {
        const e = filtered[i];
        const x = xPos(i);
        const d = e.date.getDate();
        const lbl = svgEl("text",{ x,y:PAD.top+chartH+16,"text-anchor":"middle","font-size":"10","fill":C.muted });
        lbl.textContent = d < 10 ? "0"+d : d;
        svg.appendChild(lbl);
        // вертикальная метка
        const tick = svgEl("line",{ x1:x,y1:PAD.top+chartH,x2:x,y2:PAD.top+chartH+4,stroke:C.muted2,"stroke-width":"1" });
        svg.appendChild(tick);
    }

    // ─── Линии и точки ────────────────────────────────────────────────────────
    function buildLine(valueKey, color, dashed=false) {
        const pts = filtered.map((e,i)=>({ x:xPos(i), y:yPos(e[valueKey]), v:e[valueKey], entry:e })).filter(p=>p.v>0);
        if (pts.length < 1) return;

        // Заливка под линией
        if (!dashed && pts.length > 1) {
            const areaPoints = pts.map(p=>`${p.x},${p.y}`).join(" ");
            const area = svgEl("polyline",{
                points: `${pts[0].x},${yPos(0)} ${areaPoints} ${pts[pts.length-1].x},${yPos(0)}`,
                fill: color,
                "fill-opacity":"0.07",
                stroke:"none"
            });
            svg.appendChild(area);
        }

        // Линия
        if (pts.length > 1) {
            const polyline = svgEl("polyline",{
                points: pts.map(p=>`${p.x},${p.y}`).join(" "),
                fill:"none",
                stroke: color,
                "stroke-width": dashed ? "1.5" : "2",
                "stroke-linejoin":"round",
                "stroke-linecap":"round",
                ...(dashed ? {"stroke-dasharray":"4,3"} : {})
            });
            svg.appendChild(polyline);
        }

        // Точки (кликабельные)
        for (const p of pts) {
            const dot = svgEl("circle",{
                cx: p.x,
                cy: p.y,
                r: "4",
                fill: color,
                stroke: "#111",
                "stroke-width": "1.5",
                class: "mc-dot",
                style:"cursor:pointer;"
            });

            dot.addEventListener("mouseenter", (ev) => {
                dot.setAttribute("r","6");
                const d = p.entry.date;
                const dateStr = `${d.getDate() < 10 ? "0"+d.getDate() : d.getDate()}.${d.getMonth()+1<10?"0"+(d.getMonth()+1):d.getMonth()+1}.${d.getFullYear()}`;
                tip.innerHTML = `<b style="color:${C.accentLight}">${dateStr}</b><br>Настроение: <b>${p.entry.mood > 0 ? p.entry.mood+"/10" : "—"}</b><br>Энергия: <b>${p.entry.energy > 0 ? p.entry.energy+"/10" : "—"}</b><br><span style="font-size:10px;color:#555;">Нажми чтобы открыть</span>`;
                tip.style.display = "block";
            });

            dot.addEventListener("mousemove", (ev) => {
                tip.style.left = (ev.clientX+14)+"px";
                tip.style.top  = (ev.clientY-10)+"px";
            });

            dot.addEventListener("mouseleave", () => {
                dot.setAttribute("r","4");
                tip.style.display = "none";
            });

            dot.addEventListener("click", () => {
                tip.style.display = "none";
                app.workspace.openLinkText(p.entry.path, "", false);
            });

            svg.appendChild(dot);
        }
    }

    buildLine("mood",   C.accent,   false);
    buildLine("energy", C.energy,   true);
}

redraw();

// Очистка тултипа при уходе со страницы
root.addEventListener("mouseleave", () => { tip.style.display = "none"; });
